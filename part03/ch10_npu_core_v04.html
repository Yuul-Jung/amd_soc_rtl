02전체 RTL 소스

ch10/rtl/npu_core.v · 전체 RTL
`timescale 1ns/1ps
module npu_core #(parameter integer DATA_WIDTH=16,ACC_WIDTH=40)(
    input wire clk,resetn,start_i,
    input wire [8:0] length_i,
    input wire [4*ACC_WIDTH-1:0] biases_i,
    input wire x_wr_en,w_wr_en,
    input wire [5:0] x_wr_addr,w_wr_addr,
    input wire [1:0] w_neuron,
    input wire signed [DATA_WIDTH-1:0] x_wr_data,w_wr_data,
    output wire load_ready_o,
    output reg busy_o,done_o,error_o,
    output reg [4*DATA_WIDTH-1:0] result_o,
    output reg [3:0] overflow_o
);
    localparam IDLE=0,LAUNCH=1,READ=2,WAIT_READ=3,SEND=4,WAIT_DONE=5;
    reg [2:0] state;
    reg [8:0] length_q;
    reg [5:0] index_q;
    reg [4*ACC_WIDTH-1:0] biases_q;
    wire rd_en=state==READ;
    wire signed [DATA_WIDTH-1:0] x_data;
    wire x_valid;
    wire [4*DATA_WIDTH-1:0] w_data;
    wire [3:0] w_valid;
    wire dense_ready,dense_done,dense_error;
    wire [4*DATA_WIDTH-1:0] dense_result;
    wire [3:0] dense_overflow;
    assign load_ready_o=resetn && !busy_o;
    bram_sync_buffer #(.DATA_WIDTH(DATA_WIDTH),.ADDR_WIDTH(6)) inputs_mem
      (.clk(clk),.resetn(resetn),.wr_en(x_wr_en && load_ready_o),.wr_addr(x_wr_addr),.wr_data(x_wr_data),
       .rd_en(rd_en),.rd_addr(index_q),.rd_data(x_data),.rd_valid(x_valid));
    genvar g;
    generate for(g=0;g<4;g=g+1)begin: weights_mem
        bram_sync_buffer #(.DATA_WIDTH(DATA_WIDTH),.ADDR_WIDTH(6)) mem
          (.clk(clk),.resetn(resetn),.wr_en(w_wr_en && load_ready_o && w_neuron==g),
           .wr_addr(w_wr_addr),.wr_data(w_wr_data),.rd_en(rd_en),.rd_addr(index_q),
           .rd_data(w_data[g*DATA_WIDTH+:DATA_WIDTH]),.rd_valid(w_valid[g]));
    end endgenerate
    dense_layer4 #(.DATA_WIDTH(DATA_WIDTH),.ACC_WIDTH(ACC_WIDTH)) dense
      (.clk(clk),.resetn(resetn),.start_i(state==LAUNCH),.length_i(length_q),
       .valid_i(state==SEND),.x_i(x_data),.weights_i(w_data),.biases_i(biases_q),
       .ready_o(dense_ready),.busy_o(),.done_o(dense_done),.error_o(dense_error),
       .result_o(dense_result),.overflow_o(dense_overflow));
    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;busy_o<=0;done_o<=0;error_o<=0;result_o<=0;overflow_o<=0;
            index_q<=0;length_q<=0;biases_q<=0;
        end else begin
            done_o<=0;error_o<=0;
            case(state)
                IDLE: if(start_i)begin
                    if(length_i>64 || x_wr_en || w_wr_en)error_o<=1;
                    else begin
                        length_q<=length_i;biases_q<=biases_i;index_q<=0;busy_o<=1;state<=LAUNCH;
                    end
                end
                LAUNCH: if(length_q==0)state<=WAIT_DONE;else state<=READ;
                READ: state<=WAIT_READ;
                WAIT_READ: begin
                    // 메모리 출력과 Valid가 정렬된 에지를 확인한 뒤 전송한다.
                    if(x_valid && (&w_valid))state<=SEND;
                    else begin state<=IDLE;busy_o<=0;error_o<=1;end
                end
                SEND: if(dense_ready)begin
                    if({3'b0,index_q}+9'd1==length_q)state<=WAIT_DONE;
                    else begin index_q<=index_q+1'b1;state<=READ;end
                end
                WAIT_DONE: if(dense_done)begin
                    result_o<=dense_result;overflow_o<=dense_overflow;busy_o<=0;done_o<=1;state<=IDLE;
                end
                default: begin state<=IDLE;busy_o<=0;error_o<=1;end
            endcase
            if(dense_error)begin state<=IDLE;busy_o<=0;error_o<=1;end
        end
    end
endmodule

03전체 테스트벤치

ch10/tb/tb_npu_core.sv · 전체 테스트벤치
`timescale 1ns/1ps
module tb_npu_core;
    parameter DW=16,AW=40;
    reg clk=0,rn=0,start=0,xwe=0,wwe=0;
    reg[8:0]len=0;reg[5:0]xa=0,wa=0;reg[1:0]neuron=0;
    reg signed[DW-1:0]xd=0,wd=0;
    reg[4*AW-1:0]biases=0;
    wire ready,busy,done,error;wire[4*DW-1:0]result;wire[3:0]overflow;
    npu_core #(.DATA_WIDTH(DW),.ACC_WIDTH(AW)) dut(.clk(clk),.resetn(rn),.start_i(start),.length_i(len),
      .biases_i(biases),.x_wr_en(xwe),.w_wr_en(wwe),.x_wr_addr(xa),.w_wr_addr(wa),.w_neuron(neuron),
      .x_wr_data(xd),.w_wr_data(wd),.load_ready_o(ready),.busy_o(busy),.done_o(done),.error_o(error),
      .result_o(result),.overflow_o(overflow));
    always #5 clk=~clk;
    integer f,o,n,c=0,k,i,t,run,xi,ws[0:3],expected[0:3],em;
    reg signed[AW-1:0]bs[0:3];
    reg signed[DW-1:0]xm[0:63],wm[0:3][0:63];
    string vp,ap;
    initial begin
        if(!$value$plusargs("VECTORS=%s",vp)||!$value$plusargs("ACTUAL=%s",ap))$fatal(1,"Paths");
        f=$fopen(vp,"r");o=$fopen(ap,"w");if(!f||!o)$fatal(1,"Open");
        repeat(2)@(negedge clk);rn=1;start=1;len=65;
        @(posedge clk);#1;if(!error||busy)$fatal(1,"Length guard");
        @(negedge clk);start=0;
        while(!$feof(f))begin
            if($fscanf(f,"%d %d %d %d %d %d %d %d %d %d\n",n,bs[0],bs[1],bs[2],bs[3],
              expected[0],expected[1],expected[2],expected[3],em)!=10)$fatal(1,"Header");
            for(k=0;k<n;k=k+1)begin
                if($fscanf(f,"%d %d %d %d %d\n",xi,ws[0],ws[1],ws[2],ws[3])!=5)$fatal(1,"Data");
                xm[k]=xi;for(i=0;i<4;i=i+1)wm[i][k]=ws[i];
            end
            // 역순 주소로 적재하여 전송 순서와 메모리 주소 순서를 분리한다.
            for(k=n-1;k>=0;k=k-1)for(i=0;i<4;i=i+1)begin
                @(negedge clk);if(!ready)$fatal(1,"Load ready");
                xwe=(i==0);xa=k;xd=xm[k];wwe=1;wa=k;wd=wm[i][k];neuron=i;
                @(posedge clk);#1;
            end
            @(negedge clk);xwe=0;wwe=0;
            for(run=1;run<=2;run=run+1)begin
                if(run==2)begin
                    // 계산을 시작해 중간에 취소한 뒤 재적재 없이 다시 실행한다.
                    @(negedge clk);start=1;len=n;biases=0;
                    @(negedge clk);start=0;
                    repeat(3)@(negedge clk);rn=0;
                    @(posedge clk);#1;if(busy||done||ready||result!==0)$fatal(1,"Reset cancel");
                    @(negedge clk);rn=1;
                end
                @(negedge clk);start=1;len=n;
                for(i=0;i<4;i=i+1)biases[i*AW+:AW]=bs[i];
                @(posedge clk);#1;if(!busy||done||ready)$fatal(1,"Start");
                @(negedge clk);start=0;biases=0;t=0;
                while(!done)begin
                    @(negedge clk);
                    // busy 중 주소0 오염 요청은 차단되어야 한다.
                    xwe=(t==2);wwe=(t==2);start=(t==2);xa=0;wa=0;xd=99;wd=99;neuron=0;
                    @(posedge clk);#1;t=t+1;
                    if(error)$fatal(1,"Core error state=%0d",dut.state);
                    if(t>1000)$fatal(1,"Timeout inference");
                end
                if(busy||!ready||overflow!==em[3:0])$fatal(1,"Flags");
                for(i=0;i<4;i=i+1)if($signed(result[i*DW+:DW])!==expected[i])$fatal(1,"Result dataset=%0d neuron=%0d",c+1,i);
                $fdisplay(o,"%0d %0d %0d %0d %0d %0d %0d",c+1,run,$signed(result[0*DW+:DW]),
                  $signed(result[1*DW+:DW]),$signed(result[2*DW+:DW]),$signed(result[3*DW+:DW]),overflow);
                @(negedge clk);start=0;xwe=0;wwe=0;
                @(posedge clk);#1;if(done||busy)$fatal(1,"Done pulse");
            end
            c=c+1;
        end
        if(!c)$fatal(1,"Empty");$fclose(o);$display("RTL_SIM_PASS NPU INT%0d/%0d datasets=%0d inferences=%0d",DW,AW,c,2*c);$finish;
    end
    initial begin #10000000;$fatal(1,"Timeout");end
endmodule
