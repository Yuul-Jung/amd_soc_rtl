제11장 · UART X/W/B 모델 직접 적재

PC에서 보낸 데이터가 실제 NPU X·W·Bias BRAM에 기록되는 것을 구현합니다.

실제 RTL 제작UART 데이터 경로PC 기준모델 비교

00과제 개요

기존 4부류 NPU의 모델 값을 PC에서 교체합니다. UART(비동기 직렬 통신) 한 바이트 opcode 뒤에 고정 길이 바이너리 필드를 사용합니다. 하나의 응답이 끝난 뒤 다음 명령을 보내는 stop-and-wait 방식입니다.

01전체 구성

값 변경은 uart_npu_command_controller → npu_engine4의 실제 쓰기 포트를 통해 처리됩니다.

02UART 명령·응답 표

명령뒤따르는 바이너리 바이트회신
X주소1 + signed16 빅엔디언2OK\r\n
W뉴런1 + 주소1 + signed16 2OK\r\n
B뉴런1 + signed40 빅엔디언5OK\r\n
R길이1 (1..64)Ck:4점수 또는 ERR
0~3없음고정 입력으로 분류

X/W/B는 ASCII 숫자가 아닌 **원시 바이트 값**을 전송합니다. PC에서 64개 가중치 전체가 필요한 입력은 모든 주소의 유효 상태를 먼저 채워야 합니다.

03명령 Controller 전체 RTL

mini_npu/rtl/uart_npu_command_controller.v · verilog
// (추가) UART 명령·모델 적재·추론 결과 직렬화 Controller.
// 단일 문자 '0'~'3'은 2원소 고정 표본. 11장 바이너리 명령은 아래 README 참조.
// 요청/응답은 stop-and-wait: ACK 또는 결과 LF까지 다음 명령 송신 금지.
module uart_npu_command_controller(
    input wire clk,resetn,
    input wire [7:0] rx_data_i,
    input wire rx_valid_i,
    output wire rx_ready_o,
    output wire [7:0] tx_data_o,
    output wire tx_valid_o,
    input wire tx_ready_i,
    output wire npu_start_o,
    output wire [6:0] npu_length_o,
    output wire x_wr_en_o,w_wr_en_o,b_wr_en_o,
    output reg [5:0] wr_addr_o,
    output reg [1:0] neuron_o,
    output reg signed [15:0] wr_data_o,
    output reg signed [39:0] bias_data_o,
    input wire npu_load_ready_i,
    input wire npu_busy_i,npu_done_i,npu_error_i,
    input wire [1:0] npu_class_i,
    input wire [163:0] npu_scores_i,
    output reg protocol_error_o
);
    localparam [2:0] IDLE=0,RECV=1,APPLY=2,PRESET=3,RUN=4,WAIT_NPU=5,SEND=6;
    reg [2:0] state;
    reg [7:0] opcode;
    reg [7:0] payload[0:5];
    reg [2:0] recv_index,expected_bytes;
    reg [1:0] sample_id;
    reg [3:0] preset_index;
    reg [6:0] run_length;
    reg [7:0] response[0:63];
    reg [5:0] send_length,send_index;
    integer i,j;

    assign rx_ready_o=resetn&&(state==IDLE||state==RECV);
    assign tx_valid_o=(state==SEND)&&(send_index<send_length);
    assign tx_data_o=response[send_index];
    assign npu_start_o=(state==RUN);
    assign npu_length_o=run_length;

    // (추가) 필드 형식 거부. 주소 0~63, 뉴런 0~3, 길이 1~64.
    wire accepted_fields =
        (opcode=="X" && payload[0]<64) ||
        (opcode=="W" && payload[0]<4 && payload[1]<64) ||
        (opcode=="B" && payload[0]<4) ||
        (opcode=="R" && payload[0]>=1 && payload[0]<=64);

    assign x_wr_en_o=(state==PRESET && preset_index<2 && npu_load_ready_i) ||
        (state==APPLY && opcode=="X" && accepted_fields && npu_load_ready_i);
    assign w_wr_en_o=(state==PRESET && preset_index>=2 && preset_index<10 &&
                      npu_load_ready_i) ||
        (state==APPLY && opcode=="W" && accepted_fields && npu_load_ready_i);
    assign b_wr_en_o=(state==PRESET && preset_index>=10 && npu_load_ready_i) ||
        (state==APPLY && opcode=="B" && accepted_fields && npu_load_ready_i);

    always @*begin
        wr_addr_o=0;neuron_o=0;wr_data_o=0;bias_data_o=0;
        if(state==PRESET)begin
            if(preset_index<2)begin
                wr_addr_o={2'b00,preset_index};
                // X0=(+1,+1), X1=(+1,-1), X2=(-1,+1), X3=(-1,-1)
                wr_data_o=((preset_index==0 && sample_id[1]) ||
                    (preset_index==1 && sample_id[0])) ? -16'sd1 : 16'sd1;
            end else if(preset_index<10)begin
                neuron_o=(preset_index-2)>>1;
                wr_addr_o=(preset_index-2)&1;
                wr_data_o=((neuron_o[1] && wr_addr_o==0) ||
                    (neuron_o[0] && wr_addr_o==1)) ? -16'sd1 : 16'sd1;
            end else neuron_o=preset_index-10;
        end else if(state==APPLY)begin
            case(opcode)
                "X":begin wr_addr_o=payload[0][5:0];wr_data_o={payload[1],payload[2]};end
                "W":begin neuron_o=payload[0][1:0];wr_addr_o=payload[1][5:0];
                    wr_data_o={payload[2],payload[3]};end
                "B":begin neuron_o=payload[0][1:0];
                    bias_data_o={payload[1],payload[2],payload[3],payload[4],payload[5]};end
                default:begin end
            endcase
        end
    end

    function [7:0] hexdigit(input [3:0] n);
        begin
            hexdigit=(n<10) ? (8'h30+n) : (8'h41+(n-10));
        end
    endfunction
    function [7:0] score_hex(input [40:0] score,input integer position);
        reg [43:0] ext;
        reg [3:0] nib;
        begin
            ext={{3{score[40]}},score};
            nib=(ext>>((10-position)*4)) & 4'hf;
            score_hex=hexdigit(nib);
        end
    endfunction

    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;opcode<=0;recv_index<=0;expected_bytes<=0;
            sample_id<=0;preset_index<=0;run_length<=2;
            send_length<=0;send_index<=0;protocol_error_o<=0;
        end else begin
            protocol_error_o<=0;
            case(state)
                IDLE:if(rx_valid_i && rx_ready_o)begin
                    if(rx_data_i>="0" && rx_data_i<="3")begin
                        sample_id<=rx_data_i[1:0];preset_index<=0;run_length<=2;
                        state<=PRESET;
                    end else if(rx_data_i=="X"||rx_data_i=="W"||
                        rx_data_i=="B"||rx_data_i=="R")begin
                        opcode<=rx_data_i;recv_index<=0;
                        case(rx_data_i)
                            "X":expected_bytes<=3;
                            "W":expected_bytes<=4;
                            "B":expected_bytes<=6;
                            default:expected_bytes<=1;
                        endcase
                        state<=RECV;
                    end else if(rx_data_i=="S")begin
                        response[0]<="I";response[1]<=8'h0d;response[2]<=8'h0a;
                        send_index<=0;send_length<=3;state<=SEND;
                    end else if(rx_data_i!=8'h0d && rx_data_i!=8'h0a)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end
                end
                RECV:if(rx_valid_i && rx_ready_o)begin
                    payload[recv_index]<=rx_data_i;
                    if(recv_index+3'd1==expected_bytes)state<=APPLY;
                    else recv_index<=recv_index+1'b1;
                end
                APPLY:begin
                    if(!accepted_fields || !npu_load_ready_i)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end else if(opcode=="R")begin
                        run_length<=payload[0][6:0];state<=RUN;
                    end else begin
                        response[0]<="O";response[1]<="K";
                        response[2]<=8'h0d;response[3]<=8'h0a;
                        send_index<=0;send_length<=4;state<=SEND;
                    end
                end
                PRESET:if(npu_load_ready_i)begin
                    if(preset_index==13)state<=RUN;
                    else preset_index<=preset_index+1'b1;
                end
                RUN:state<=WAIT_NPU;
                WAIT_NPU:begin
                    if(npu_error_i)begin
                        protocol_error_o<=1;
                        response[0]<="E";response[1]<="R";response[2]<="R";
                        response[3]<=8'h0d;response[4]<=8'h0a;
                        send_index<=0;send_length<=5;state<=SEND;
                    end else if(npu_done_i)begin
                        // Cx: + 4 signed 41bit scores each in 11 hex chars + CRLF.
                        response[0]<="C";response[1]<=8'h30+{6'b0,npu_class_i};
                        response[2]<=":";
                        for(i=0;i<4;i=i+1)begin
                            for(j=0;j<11;j=j+1)
                                response[3+i*12+j]<=score_hex(
                                    npu_scores_i[i*41+:41],j);
                            if(i<3)response[14+i*12]<=",";
                        end
                        response[50]<=8'h0d;response[51]<=8'h0a;
                        send_index<=0;send_length<=52;state<=SEND;
                    end
                end
                SEND:if(tx_valid_o && tx_ready_i)begin
                    if(send_index+6'd1==send_length)begin
                        send_index<=0;state<=IDLE;
                    end else send_index<=send_index+1'b1;
                end
                default:begin state<=IDLE;protocol_error_o<=1;end
            endcase
        end
    end
endmodule

04실제 UART 통합 TB

mini_npu/tb/tb_npu_uart.sv · systemverilog
// (추가) CPU/AXI 없이 실제 UART RX 핀→BRAM→DSP 4PE→UART TX 핀 파형 검증.
`timescale 1ns/1ps
module tb_npu_uart;
    localparam integer CPB=868;
    reg clk=0,btnC=1,uart_rxd=1;
    wire uart_txd,dp;
    wire [15:0] led;
    wire [6:0] seg;
    wire [3:0] an;
    wire [7:0] byte_out;
    wire byte_valid,fe,ov;
    integer c_count=0,ack_count=0,err_count=0,nbytes=0;
    integer k,i,j;
    reg [7:0] chars[0:63];
    reg [8*256-1:0] csvpath;
    integer csv_file;
    always #5 clk=~clk;
    basys3_npu_top dut(.clk(clk),.btnC(btnC),
        .uart_rxd(uart_rxd),.uart_txd(uart_txd),
        .led(led),.seg(seg),.an(an),.dp(dp));
    uart_rx tx_monitor(.clk(clk),.resetn(!btnC),.rx_i(uart_txd),
        .out_data_o(byte_out),.out_valid_o(byte_valid),
        .out_ready_i(1'b1),.framing_error_o(fe),.overrun_o(ov));

    task automatic send_byte(input [7:0] b);
        integer bitno;
        begin
            @(negedge clk);uart_rxd=0;repeat(CPB)@(negedge clk);
            for(bitno=0;bitno<8;bitno=bitno+1)begin
                uart_rxd=b[bitno];repeat(CPB)@(negedge clk);
            end
            uart_rxd=1;repeat(CPB)@(negedge clk);
        end
    endtask
    function automatic [7:0] hexchar(input [3:0] value);
        begin
            if(value<10)hexchar="0"+value;
            else hexchar="A"+(value-10);
        end
    endfunction
    function automatic signed [40:0] golden(input integer s,input integer n);
        integer x0,x1,w0,w1;
        begin
            x0=(s&2)?-1:1; x1=(s&1)?-1:1;
            w0=(n&2)?-1:1; w1=(n&1)?-1:1;
            golden=x0*w0+x1*w1;
            if(c_count==5 && n==3)golden=golden+10;
        end
    endfunction
    function automatic [7:0] gold_hex(input signed [40:0] val,input integer pos);
        reg [43:0] ext;
        begin
            ext={{3{val[40]}},val};
            gold_hex=hexchar((ext>>((10-pos)*4))&4'hf);
        end
    endfunction
    // (추가) 실제 TX 라인에서 복조한 ASCII HEX를 signed 41-bit 값으로 변환.
    // CSV에 골든값을 복사하면 UART 검증 증거가 되지 않으므로 디코딩값을 기록.
    function automatic signed [40:0] decoded_score(input integer lane);
        integer p;
        reg [7:0] ch;
        reg [3:0] hex;
        reg [43:0] packed_hex;
        begin
            packed_hex=44'd0;
            for(p=0;p<11;p=p+1)begin
                ch=chars[3+lane*12+p];
                if(ch>="0" && ch<="9")hex=ch-"0";
                else if(ch>="A" && ch<="F")hex=ch-"A"+10;
                else $fatal(1,"Non-hex RTL UART output");
                packed_hex=(packed_hex<<4)|hex;
            end
            decoded_score=$signed(packed_hex[40:0]);
        end
    endfunction

    integer scenario,exp_cls;
    reg signed [40:0] exp_score;
    always @(posedge clk)if(byte_valid)begin
        if(nbytes>=63)$fatal(1,"TX line overflow");
        if(byte_out!=8'h0a)begin
            chars[nbytes]=byte_out;nbytes=nbytes+1;
        end else begin
            if(nbytes==51 && chars[0]=="C")begin
                scenario=(c_count<4)?c_count:0;
                exp_cls=(c_count==5)?3:scenario;
                if(chars[1]!==("0"+exp_cls))$fatal(1,"UART CLASS mismatch");
                if(chars[2]!==":")$fatal(1,"Bad UART result header");
                // (추가) 응답 점수 구분자와 CR도 실제 비트 복조값으로 검증.
                if(chars[14]!=="," || chars[26]!=="," || chars[38]!=="," ||
                   chars[50]!==8'h0d)$fatal(1,"UART score separators or CR mismatch");
                for(i=0;i<4;i=i+1)begin
                    exp_score=golden(scenario,i);
                    for(j=0;j<11;j=j+1)
                        if(chars[3+i*12+j]!==gold_hex(exp_score,j))
                            $fatal(1,"UART SCORE mismatch case=%0d lane=%0d hex=%0d",c_count,i,j);
                end
                if(led[exp_cls]!==1'b1)$fatal(1,"LED class mismatch");
                // (변경) 예상값이 아니라 TX 비트파형에서 실제 수신한 4개 점수 기록.
                $fdisplay(csv_file,"%0d,%0d,%0d,%0d,%0d,%0d",c_count,exp_cls,
                    decoded_score(0),decoded_score(1),
                    decoded_score(2),decoded_score(3));
                c_count=c_count+1;
            end else if(nbytes==3 && chars[0]=="O" && chars[1]=="K")begin
                ack_count=ack_count+1;
            end else if(nbytes==4 && chars[0]=="E" && chars[1]=="R" && chars[2]=="R")begin
                err_count=err_count+1;
            end else $fatal(1,"Unexpected UART response len=%0d",nbytes+1);
            nbytes=0;
        end
    end
    initial begin
        csvpath="part03/mini_npu/outputs/predictions.csv";
        if($value$plusargs("CSV=%s",csvpath))begin end
        csv_file=$fopen(csvpath,"w");
        if(csv_file==0)$fatal(1,"Cannot open CSV file");
        $fdisplay(csv_file,"scenario,class,score0,score1,score2,score3");
        repeat(10)@(negedge clk);btnC=0;
        for(k=0;k<4;k=k+1)begin
            send_byte("0"+k);
            wait(c_count==k+1);
        end
        send_byte("0");wait(c_count==5);
        // 11장: Bias3=+10으로 실제 모델 데이터를 변경한 뒤 재추론.
        send_byte("B");send_byte(8'h03);
        send_byte(0);send_byte(0);send_byte(0);send_byte(0);send_byte(10);
        wait(ack_count==1);
        send_byte("R");send_byte(8'h02);
        wait(c_count==6);
        // 미적재 3번째 입력을 요청하면 ERR를 반환해야 한다.
        send_byte("R");send_byte(8'h03);
        wait(err_count==1);
        if(fe||ov)$fatal(1,"UART line framing/overrun");
        $fclose(csv_file);
        $display("INTEGRATED_UART_RTL_SIM_PASS classes=6 model_edit=1 invalid_length=1");
        $finish;
    end
    initial begin #100000000;$fatal(1,"UART NPU end-to-end timeout");end
endmodule

주의 · 명령 Timeout 없음: Controller는 명령 바이트 뒤 정해진 개수의 페이로드를 받을 때까지 기다립니다. 예를 들어 B 뒤 2바이트만 보내고 멈추면 응답이 없고, 이어서 보낸 바이트는 남은 페이로드로 소비됩니다. 이때는 btnC Reset으로 복구합니다. PC는 반드시 응답 LF를 받은 뒤 다음 명령을 보냅니다.

잘못된 Opcode·주소·뉴런·길이(ERR), 경계값 63(OK), CR/LF 무시, 명령 중단 동작은 bash part03/mini_npu/scripts/run_adversarial.sh의 통합 시험으로 확인합니다.

06실습 확인

OK는 유효한 X/W/B 수락, ERR는 잘못된 길이·주소 등에서 발생합니다. 실제 송신 TX 비트파형을 디코딩하여 결과를 비교하는 TB를 사용합니다.
