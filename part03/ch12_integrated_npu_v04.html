제12장 · UART·BRAM·DSP·4PE 소형 NPU 통합

실제 메모리 데이터를 DSP에서 곱하고 4뉴런 결과를 UART·LED·FND로 표시합니다.

실제 RTL 제작UART 데이터 경로PC 기준모델 비교

00종합 과제 개요

PC 터미널의 문자 0~3 또는 X/W/B/R 데이터가 입력되면 X/W BRAM → DSP 등록 곱셈 → 누산 → 부호 있는 41비트 점수 → Argmax4의 실제 연산 경로를 거칩니다.

01Top 연결 구조

Top에는 연산·우선순위·상태 FSM을 두지 않고 uart_core, uart_npu_command_controller, npu_engine4, npu_display_controller를 연결합니다.

// (추가) 02장 등록 DSP 곱을 다음 에지에 누산한다. 곱을 다시 수행하지 않는다.
module npu_dsp_mac_lane #(
    parameter integer DATA_WIDTH=16,
    parameter integer ACC_WIDTH=40
)(
    input wire clk,resetn,clear_i,valid_i,
    input wire signed [DATA_WIDTH-1:0] x_i,w_i,
    output reg signed [ACC_WIDTH-1:0] acc_o
);
    wire signed [2*DATA_WIDTH-1:0] product;
    wire product_valid;
    dsp_multiplier #(.A_WIDTH(DATA_WIDTH),.B_WIDTH(DATA_WIDTH)) dsp(
        .clk(clk),.resetn(resetn),.clear_i(clear_i),
        .valid_i(valid_i),.a_i(x_i),.b_i(w_i),
        .product_o(product),.valid_o(product_valid));
    wire signed [ACC_WIDTH-1:0] product_extended =
        {{(ACC_WIDTH-2*DATA_WIDTH){product[2*DATA_WIDTH-1]}},product};
    always @(posedge clk)begin
        if(!resetn || clear_i)acc_o<=0;
        else if(product_valid)acc_o<=acc_o+product_extended;
    end
    // synthesis translate_off
    initial if(ACC_WIDTH<2*DATA_WIDTH)$fatal(1,"ACC_WIDTH too small");
    // synthesis translate_on
endmodule

// (추가) 04장 BRAM 구조 + 02장 DSP 레지스터 곱 + 4개 누산기 통합.
// 12장 최종 경로는 기존 ch07 벡터 닷의 곱 경로를 중복 사용하지 않는다.
// 메모리 쓰기는 IDLE에서만 허용하며 Reset 시 유효 비트만 지운다.
module npu_engine4(
    input wire clk,resetn,start_i,
    input wire [6:0] length_i,
    input wire x_wr_en,w_wr_en,b_wr_en,
    input wire [5:0] addr_i,
    input wire [1:0] neuron_i,
    input wire signed [15:0] data_i,
    input wire signed [39:0] bias_i,
    output wire load_ready_o,
    output reg busy_o,done_o,error_o,
    output reg [1:0] class_o,
    output reg [163:0] scores_o
);
    localparam [2:0] IDLE=0,READ=1,ISSUE=2,DRAIN1=3,DRAIN2=4;
    reg [2:0] state;
    reg [6:0] length_q;
    reg [5:0] index_q;
    (* ram_style="block" *) reg signed [15:0] xmem[0:63];
    (* ram_style="block" *) reg signed [15:0] w0[0:63];
    (* ram_style="block" *) reg signed [15:0] w1[0:63];
    (* ram_style="block" *) reg signed [15:0] w2[0:63];
    (* ram_style="block" *) reg signed [15:0] w3[0:63];
    reg [63:0] x_seen,w0_seen,w1_seen,w2_seen,w3_seen;
    reg signed [39:0] bias0,bias1,bias2,bias3;
    reg signed [15:0] x_sample,w_sample0,w_sample1,w_sample2,w_sample3;

    reg input_valid;
    integer k;
    always @* begin
        input_valid=(length_i!=0 && length_i<=64);
        for(k=0;k<64;k=k+1)
            if(k<length_i &&
               (!x_seen[k] || !w0_seen[k] || !w1_seen[k] ||
                !w2_seen[k] || !w3_seen[k]))
                input_valid=0;
    end
    wire any_write=x_wr_en||w_wr_en||b_wr_en;
    wire launch=(state==IDLE)&&start_i&&input_valid&&!any_write;
    wire mul_valid=(state==ISSUE);
    assign load_ready_o=resetn&&!busy_o&&!start_i;

    wire signed [39:0] acc0,acc1,acc2,acc3;
    npu_dsp_mac_lane lane0(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample0),.acc_o(acc0));
    npu_dsp_mac_lane lane1(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample1),.acc_o(acc1));
    npu_dsp_mac_lane lane2(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample2),.acc_o(acc2));
    npu_dsp_mac_lane lane3(.clk(clk),.resetn(resetn),.clear_i(launch),
        .valid_i(mul_valid),.x_i(x_sample),.w_i(w_sample3),.acc_o(acc3));
    // 41비트 signed: 40비트 누산값에 Bias를 더할 때 부호 확장 1비트.
    wire signed [40:0] score0=$signed({acc0[39],acc0})+$signed({bias0[39],bias0});
    wire signed [40:0] score1=$signed({acc1[39],acc1})+$signed({bias1[39],bias1});
    wire signed [40:0] score2=$signed({acc2[39],acc2})+$signed({bias2[39],bias2});
    wire signed [40:0] score3=$signed({acc3[39],acc3})+$signed({bias3[39],bias3});
    reg [1:0] best_class;
    reg signed [40:0] best_score;
    always @*begin
        best_class=0;best_score=score0;
        if(score1>best_score)begin best_score=score1;best_class=1;end
        if(score2>best_score)begin best_score=score2;best_class=2;end
        if(score3>best_score)begin best_score=score3;best_class=3;end
    end

    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;busy_o<=0;done_o<=0;error_o<=0;
            class_o<=0;scores_o<=0;length_q<=0;index_q<=0;
            x_seen<=0;w0_seen<=0;w1_seen<=0;w2_seen<=0;w3_seen<=0;
            bias0<=0;bias1<=0;bias2<=0;bias3<=0;
            x_sample<=0;w_sample0<=0;w_sample1<=0;w_sample2<=0;w_sample3<=0;
        end else begin
            done_o<=0;error_o<=0;
            if(load_ready_o)begin
                if(x_wr_en)begin
                    xmem[addr_i]<=data_i;x_seen[addr_i]<=1;
                end
                if(w_wr_en)case(neuron_i)
                    0:begin w0[addr_i]<=data_i;w0_seen[addr_i]<=1;end
                    1:begin w1[addr_i]<=data_i;w1_seen[addr_i]<=1;end
                    2:begin w2[addr_i]<=data_i;w2_seen[addr_i]<=1;end
                    3:begin w3[addr_i]<=data_i;w3_seen[addr_i]<=1;end
                endcase
                if(b_wr_en)case(neuron_i)
                    0:bias0<=bias_i;
                    1:bias1<=bias_i;
                    2:bias2<=bias_i;
                    3:bias3<=bias_i;
                endcase
            end else if(any_write)error_o<=1;
            case(state)
                IDLE:if(start_i)begin
                    if(!input_valid||any_write)error_o<=1;
                    else begin
                        state<=READ;length_q<=length_i;index_q<=0;busy_o<=1;
                    end
                end
                READ:begin
                    x_sample<=xmem[index_q];
                    w_sample0<=w0[index_q];w_sample1<=w1[index_q];
                    w_sample2<=w2[index_q];w_sample3<=w3[index_q];
                    state<=ISSUE;
                end
                ISSUE:begin
                    if({1'b0,index_q}+7'd1==length_q)state<=DRAIN1;
                    else begin index_q<=index_q+1'b1;state<=READ;end
                end
                DRAIN1:state<=DRAIN2;
                DRAIN2:begin
                    scores_o<={score3,score2,score1,score0};
                    class_o<=best_class;done_o<=1;busy_o<=0;state<=IDLE;
                end
                default:begin state<=IDLE;busy_o<=0;error_o<=1;end
            endcase
        end
    end
endmodule

// (추가) FND와 LED 표현 정책은 Top이 아닌 Controller에서 처리.
module npu_display_controller(
    input wire clk,resetn,
    input wire [1:0] class_i,
    input wire busy_i,done_i,error_i,protocol_error_i,
    input wire uart_framing_error_i,uart_overrun_i,
    output wire [15:0] led_o,
    output reg [6:0] seg_o,
    output reg [3:0] an_o,
    output wire dp_o
);
    reg done_latched,error_latched;
    reg [17:0] scan_count;
    reg [3:0] digit;
    wire [3:0] onehot=4'b0001<<class_i;
    always @(posedge clk)begin
        if(!resetn)begin
            done_latched<=0;error_latched<=0;scan_count<=0;
        end else begin
            scan_count<=scan_count+1'b1;
            if(done_i)done_latched<=1;
            if(error_i||protocol_error_i||uart_framing_error_i||uart_overrun_i)
                error_latched<=1;
        end
    end
    assign led_o[3:0]=onehot;
    assign led_o[12:4]=9'd0;
    assign led_o[13]=busy_i;
    assign led_o[14]=done_latched;
    assign led_o[15]=error_latched;
    assign dp_o=1'b1;
    always @*begin
        case(scan_count[17:16])
            2'd0:begin an_o=4'b1110;digit={2'b00,class_i};end
            2'd1:begin an_o=4'b1101;digit=0;end
            2'd2:begin an_o=4'b1011;digit=0;end
            default:begin an_o=4'b0111;digit=0;end
        endcase
        // Active low: segments {g,f,e,d,c,b,a} (seg[0] is a).
        case(digit)
            0:seg_o=7'b1000000;
            1:seg_o=7'b1111001;
            2:seg_o=7'b0100100;
            3:seg_o=7'b0110000;
            default:seg_o=7'b1111111;
        endcase
    end
endmodule

// (추가) Basys3 btnC 비동기 Assertion, 동기 Release Reset.
module npu_reset_controller(
    input wire clk,btnC,
    output wire resetn
);
    reg [1:0] release_q;
    always @(posedge clk or posedge btnC)
        if(btnC)release_q<=2'b00;
        else release_q<={release_q[0],1'b1};
    assign resetn=release_q[1];
endmodule

// (추가) 실제 DSP 등록 곱→BRAM→4PE 누적 NPU 정수 독립 검증.
`timescale 1ns/1ps
module tb_npu_engine4;
    reg clk=0,resetn=0,start=0,xwe=0,wwe=0,bwe=0;
    reg [6:0] len=2;
    reg [5:0] addr=0;
    reg [1:0] neuron=0;
    reg signed [15:0] data=0;
    reg signed [39:0] bias=0;
    wire load_ready,busy,done,error;
    wire [1:0] cls;
    wire [163:0] scores;
    integer sample,k,n;
    always #5 clk=~clk;
    npu_engine4 dut(.clk(clk),.resetn(resetn),.start_i(start),.length_i(len),
        .x_wr_en(xwe),.w_wr_en(wwe),.b_wr_en(bwe),.addr_i(addr),
        .neuron_i(neuron),.data_i(data),.bias_i(bias),
        .load_ready_o(load_ready),.busy_o(busy),.done_o(done),.error_o(error),
        .class_o(cls),.scores_o(scores));
    task automatic write_x(input [5:0] a,input signed [15:0] v);
        begin @(negedge clk);xwe=1;addr=a;data=v;
              @(negedge clk);xwe=0;end
    endtask
    task automatic write_w(input [1:0] nn,input [5:0] a,input signed [15:0] v);
        begin @(negedge clk);wwe=1;neuron=nn;addr=a;data=v;
              @(negedge clk);wwe=0;end
    endtask
    task automatic write_b(input [1:0] nn,input signed [39:0] v);
        begin @(negedge clk);bwe=1;neuron=nn;bias=v;
              @(negedge clk);bwe=0;end
    endtask
    task automatic run_npu(input [6:0] count);
        begin @(negedge clk);len=count;start=1;
              @(negedge clk);start=0;
              wait(done || error);#1;end
    endtask
    function automatic signed [40:0] exp_score(input integer s,input integer row);
        integer x0,x1,w0,w1;
        begin
            x0=(s&2)?-1:1;x1=(s&1)?-1:1;
            w0=(row&2)?-1:1;w1=(row&1)?-1:1;
            exp_score=x0*w0+x1*w1;
        end
    endfunction
    initial begin
        repeat(8)@(negedge clk);resetn=1;
        for(n=0;n<4;n=n+1)begin
            write_w(n,0,(n&2)?-16'sd1:16'sd1);
            write_w(n,1,(n&1)?-16'sd1:16'sd1);
        end
        for(sample=0;sample<4;sample=sample+1)begin
            write_x(0,(sample&2)?-16'sd1:16'sd1);
            write_x(1,(sample&1)?-16'sd1:16'sd1);
            run_npu(2);
            if(error) $fatal(1,"Unexpected NPU error sample=%0d",sample);
            if(cls!==sample[1:0])$fatal(1,"Class %0d != sample %0d",cls,sample);
            for(k=0;k<4;k=k+1)
                if($signed(scores[k*41+:41])!==exp_score(sample,k))
                    $fatal(1,"Score mismatch s=%0d lane=%0d",sample,k);
        end
        write_x(0,1);write_x(1,1);write_b(3,10);
        run_npu(2);
        if(cls!==2'd3 || $signed(scores[3*41+:41])!==41'sd8)
            $fatal(1,"Bias update ignored");
        @(negedge clk);len=3;start=1;
        @(negedge clk);start=0;
        wait(error);#1;
        $display("NPU_ENGINE_RTL_SIM_PASS samples=4 bias_update=1 invalid_len=1");
        $finish;
    end
    initial begin #1000000;$fatal(1,"Engine TB timeout");end
endmodule

        .out_ready_i(1'b1),.framing_error_o(fe),.overrun_o(ov));

    task automatic send_byte(input [7:0] b);
        integer bitno;
        begin
            @(negedge clk);uart_rxd=0;repeat(CPB)@(negedge clk);
            for(bitno=0;bitno<8;bitno=bitno+1)begin
                uart_rxd=b[bitno];repeat(CPB)@(negedge clk);
            end
            uart_rxd=1;repeat(CPB)@(negedge clk);
        end
    endtask
    function automatic [7:0] hexchar(input [3:0] value);
        begin
            if(value<10)hexchar="0"+value;
            else hexchar="A"+(value-10);
        end
    endfunction
    function automatic signed [40:0] golden(input integer s,input integer n);
        integer x0,x1,w0,w1;
        begin
            x0=(s&2)?-1:1; x1=(s&1)?-1:1;
            w0=(n&2)?-1:1; w1=(n&1)?-1:1;
            golden=x0*w0+x1*w1;
            if(c_count==5 && n==3)golden=golden+10;
        end
    endfunction
    function automatic [7:0] gold_hex(input signed [40:0] val,input integer pos);
        reg [43:0] ext;
        begin
            ext={{3{val[40]}},val};
            gold_hex=hexchar((ext>>((10-pos)*4))&4'hf);
        end
    endfunction
    // (추가) 실제 TX 라인에서 복조한 ASCII HEX를 signed 41-bit 값으로 변환.
    // CSV에 골든값을 복사하면 UART 검증 증거가 되지 않으므로 디코딩값을 기록.
    function automatic signed [40:0] decoded_score(input integer lane);
        integer p;
        reg [7:0] ch;
        reg [3:0] hex;
        reg [43:0] packed_hex;
        begin
            packed_hex=44'd0;
            for(p=0;p<11;p=p+1)begin
                ch=chars[3+lane*12+p];
                if(ch>="0" && ch<="9")hex=ch-"0";
                else if(ch>="A" && ch<="F")hex=ch-"A"+10;
                else $fatal(1,"Non-hex RTL UART output");
                packed_hex=(packed_hex<<4)|hex;
            end
            decoded_score=$signed(packed_hex[40:0]);
        end
    endfunction

    integer scenario,exp_cls;
    reg signed [40:0] exp_score;
    always @(posedge clk)if(byte_valid)begin
        if(nbytes>=63)$fatal(1,"TX line overflow");
        if(byte_out!=8'h0a)begin
            chars[nbytes]=byte_out;nbytes=nbytes+1;
        end else begin
            if(nbytes==51 && chars[0]=="C")begin
                scenario=(c_count<4)?c_count:0;
                exp_cls=(c_count==5)?3:scenario;
                if(chars[1]!==("0"+exp_cls))$fatal(1,"UART CLASS mismatch");
                if(chars[2]!==":")$fatal(1,"Bad UART result header");
                // (추가) 응답 점수 구분자와 CR도 실제 비트 복조값으로 검증.
                if(chars[14]!=="," || chars[26]!=="," || chars[38]!=="," ||
                   chars[50]!==8'h0d)$fatal(1,"UART score separators or CR mismatch");
                for(i=0;i<4;i=i+1)begin
                    exp_score=golden(scenario,i);
                    for(j=0;j<11;j=j+1)
                        if(chars[3+i*12+j]!==gold_hex(exp_score,j))
                            $fatal(1,"UART SCORE mismatch case=%0d lane=%0d hex=%0d",c_count,i,j);
                end
                if(led[exp_cls]!==1'b1)$fatal(1,"LED class mismatch");
                // (변경) 예상값이 아니라 TX 비트파형에서 실제 수신한 4개 점수 기록.
                $fdisplay(csv_file,"%0d,%0d,%0d,%0d,%0d,%0d",c_count,exp_cls,
                    decoded_score(0),decoded_score(1),
                    decoded_score(2),decoded_score(3));
                c_count=c_count+1;
            end else if(nbytes==3 && chars[0]=="O" && chars[1]=="K")begin
                ack_count=ack_count+1;
            end else if(nbytes==4 && chars[0]=="E" && chars[1]=="R" && chars[2]=="R")begin
                err_count=err_count+1;
            end else $fatal(1,"Unexpected UART response len=%0d",nbytes+1);
            nbytes=0;
        end
    end
    initial begin
        csvpath="part03/mini_npu/outputs/predictions.csv";
        if($value$plusargs("CSV=%s",csvpath))begin end
        csv_file=$fopen(csvpath,"w");
        if(csv_file==0)$fatal(1,"Cannot open CSV file");
        $fdisplay(csv_file,"scenario,class,score0,score1,score2,score3");
        repeat(10)@(negedge clk);btnC=0;
        for(k=0;k<4;k=k+1)begin
            send_byte("0"+k);
            wait(c_count==k+1);
        end
        send_byte("0");wait(c_count==5);
        // 11장: Bias3=+10으로 실제 모델 데이터를 변경한 뒤 재추론.
        send_byte("B");send_byte(8'h03);
        send_byte(0);send_byte(0);send_byte(0);send_byte(0);send_byte(10);
        wait(ack_count==1);
        send_byte("R");send_byte(8'h02);
        wait(c_count==6);
        // 미적재 3번째 입력을 요청하면 ERR를 반환해야 한다.
        send_byte("R");send_byte(8'h03);
        wait(err_count==1);
        if(fe||ov)$fatal(1,"UART line framing/overrun");
        $fclose(csv_file);
        $display("INTEGRATED_UART_RTL_SIM_PASS classes=6 model_edit=1 invalid_length=1");
        $finish;
    end
    initial begin #100000000;$fatal(1,"UART NPU end-to-end timeout");end
endmodule

#!/usr/bin/env python3
"""12장: HDL이 기록한 실제 예측 CSV를 독립 정수 정답과 비교한다."""
import csv
import sys
from pathlib import Path


def expected_row(scenario):
    s = scenario if scenario < 4 else 0
    x = (-1 if s & 2 else 1, -1 if s & 1 else 1)
    values = []
    for lane in range(4):
        w = (-1 if lane & 2 else 1, -1 if lane & 1 else 1)
        dot = x[0] * w[0] + x[1] * w[1]
        values.append(dot + (10 if scenario == 5 and lane == 3 else 0))
    best = max(range(4), key=lambda k: (values[k], -k))
    return [scenario, best, *values]


def main():
    path = Path(sys.argv[1])
    with path.open(newline="", encoding="utf-8") as stream:
        reader = csv.reader(stream)
        assert next(reader) == [
            "scenario", "class", "score0", "score1", "score2", "score3"
        ]
        rows = [[int(v) for v in row] for row in reader]
    expected = [expected_row(s) for s in range(6)]
    if rows != expected:
        for scenario, (got, want) in enumerate(zip(rows, expected)):
            if got != want:
                print(f"FAIL scenario {scenario}: got={got} expected={want}")
        raise SystemExit("GOLDEN_COMPARE_FAIL")
    print("PC_GOLDEN_COMPARE_PASS predictions=6 model_edit=1")


if __name__ == "__main__":
    main()
