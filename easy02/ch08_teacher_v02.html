제2부 08장 · 교수용 상세 해설

원리·배경: 아날로그 전압은 0/1이 아닌 연속적인 값입니다. XADC Wizard가 전압을 디지털 코드로 바꾸면 CPU가 수치 범위와 PWM 밝기 비율을 계산합니다. 12비트는 0~4095개의 코드 값 표현을 의미합니다.

실습 전체 동작: JXADC VAUX6 차동 아날로그 입력 → AMD XADC Wizard → AXI 레지스터 → C 코드의 ADC 변환·정책 → PWM duty → LED로 전달합니다.

전기·타이밍·구현 제약: XADC 핀의 실제 허용 전압 범위와 분압·센서 전원을 공식 보드 문서와 대조합니다. 전체 C 코드의 수치 마스킹·시프트·주소 오프셋을 빠뜨리지 않습니다.

1. 기능 개요와 기존 장 연계

목표: Basys3 JXADC VAUX6의 아날로그 전압을 읽고, UART ADC READ로 확인한 뒤 ADC PWM으로 기존 LED1 PWM Duty를 한 번 갱신한다.

이 장은 신규 사용자 ADC RTL 제작이 아니라 AMD 제공 XADC Wizard IP 재사용 과제이다. 제2부 01~07장의 검증된 기능·기본 코드는 변경하지 않는다.

static void uart_puts(const char *s)
{
    while (*s) XUartLite_SendByte(UART_BASEADDR, (u8)*s++);
}
static void uart_hex32(u32 v)
{
    const char hex[] = "0123456789ABCDEF";
    int shift;
    for (shift = 28; shift >= 0; shift -= 4)
        XUartLite_SendByte(UART_BASEADDR, (u8)hex[(v >> shift) & 15U]);
}
static void uart_dec4(u32 value)
{
    int d;
    u32 divisors[4] = {1000U, 100U, 10U, 1U};
    for (d = 0; d < 4; ++d) {
        u32 n = (value / divisors[d]) % 10U;
        XUartLite_SendByte(UART_BASEADDR, (u8)('0' + n));
    }
}

// (추가) 0~4294967295를 선행 0 없이 UART에 출력.
static void uart_dec_u32(u32 value)
{
    char tmp[10]; // u32 십진수 최대 10자리.
    unsigned used = 0U;
    do {
        tmp[used++] = (char)('0' + (value % 10U));
        value /= 10U;
    } while (value != 0U);
    while (used != 0U)
        XUartLite_SendByte(UART_BASEADDR, (u8)tmp[--used]);
}

// (추가) 이전 설정이 반영 중이면 BUSY, 아니면 다음 주기 적용을 예약.
static int pwm_queue_percent(u8 percent)
{
    u32 duty;
    if ((Xil_In32(P2_04_PWM_BASEADDR + PWM_STATUS) & 1U) != 0U)
        return 0; // 기존 Pending 설정을 덮지 않는다.
    duty = pwm_duty_for_percent(PWM_PERIOD_TICKS, percent);
    Xil_Out32(P2_04_PWM_BASEADDR + PWM_PERIOD_CFG, PWM_PERIOD_TICKS);
    Xil_Out32(P2_04_PWM_BASEADDR + PWM_DUTY_CFG, duty);
    Xil_Out32(P2_04_PWM_BASEADDR + PWM_CTRL_CFG, percent ? 1U : 0U);
    Xil_Out32(P2_04_PWM_BASEADDR + PWM_APPLY, 1U);
    pwm_requested_percent = percent;
    return 1;
}

// (추가) 06장 STATUS는 최근 완료된 샘플(VALID/TIMEOUT/RANGE)을 보존.
// SEQ is a lightweight change marker; the HW has no atomic multiword latch.
static void sonic_read_state(u32 *status, u32 *cm, u32 *seq)
{
    u32 a, b;
    unsigned tries;
    for (tries = 0; tries < 3U; ++tries) {
        a = Xil_In32(P2_06_SONIC_BASEADDR + SONIC_SEQ);
        *status = Xil_In32(P2_06_SONIC_BASEADDR + SONIC_STATUS);
        *cm = Xil_In32(P2_06_SONIC_BASEADDR + SONIC_DIST_CM);
        b = Xil_In32(P2_06_SONIC_BASEADDR + SONIC_SEQ);
        if (a == b) { *seq = b; return; }
    }
    *seq = b; // Contended snapshot: STATUS logs must not claim atomicity.
}

static void sonic_commit_ctrl(void)
{
    Xil_Out32(P2_06_SONIC_BASEADDR + SONIC_CTRL,
              (u32)sonic_auto_enabled | ((u32)sonic_warn_enabled << 2));
}

static void sonic_print_reading(void)
{
    u32 st, cm, seq, echo_us;

    uart_puts("SONIC CM="); uart_dec_u32(cm);

    uart_puts(" TIMEOUT="); uart_dec_u32((st >> 2) & 1U);

}

static void fnd_commit(void)
{
    if (sonic_view) {
        u32 st, cm, seq;
        uint16_t visible;
        sonic_read_state(&st,&cm,&seq);
        visible = sonic_visible_cm(st,cm);
        // (추가) D3..D0: decimal 0002..0400 cm; no unit/dot hardware.
        // Invalid/timeout => FND blank, not a misleading '0000'.
        Xil_Out32(P2_02_FND_BASEADDR + FND_VAL_REG, p2_to_bcd4(visible));
        Xil_Out32(P2_02_FND_BASEADDR + FND_CTRL_REG,
                  (sonic_view_fresh && sonic_sample_valid(st,cm)) ? 1U : 0U);
        return;
    }
    // (유지) P2-03 VALUE/EVENTS shown when SONIC OFF.
    Xil_Out32(P2_02_FND_BASEADDR + FND_VAL_REG,
              p2_to_bcd4(p2_visible_decimal(&app)));
    Xil_Out32(P2_02_FND_BASEADDR + FND_CTRL_REG, (u32)app.display_on);
}

static void service_sonic_display(void)

    if (seq != sonic_seq_seen) {
        sonic_seq_seen = seq;
        if (sonic_view) { sonic_view_fresh = 1U; fnd_commit(); }
    }
}

// Level interrupt: read PENDING, acknowledge W1C, count each asserted type.
static void btn_irq_handler(void *unused)
{
    u32 pending;
    unsigned i;
    (void)unused;
    pending = Xil_In32(P2_03_BTN_BASEADDR + BTN_PENDING) & 0xFFU;
    if (!pending) return;
    Xil_Out32(P2_03_BTN_BASEADDR + BTN_PENDING, pending); // W1C
    for (i = 0; i < 8U; ++i)
        if (pending & (1U << i)) irq_counts[i]++;
}

static int setup_button_interrupt(void)
{
    int status;
#ifdef SDT
    // The physical interrupt ID and parent are project-specific.
    status = XSetupInterruptSystem(0, (void *)btn_irq_handler,
        P2_03_BTN_INTR_ID, (UINTPTR)P2_03_BTN_INTR_PARENT,
        XINTERRUPT_DEFAULT_PRIORITY);
#else
    status = XIntc_Initialize(&intc, P2_03_INTC_DEVICE_ID);
    if (status != XST_SUCCESS) return status;
    status = XIntc_Connect(&intc, P2_03_BTN_INTR_ID,
        (XInterruptHandler)btn_irq_handler, 0);
    if (status != XST_SUCCESS) return status;
    status = XIntc_Start(&intc, XIN_REAL_MODE);
    if (status != XST_SUCCESS) return status;
    XIntc_Enable(&intc, P2_03_BTN_INTR_ID);
    Xil_ExceptionInit();
    Xil_ExceptionRegisterHandler(XIL_EXCEPTION_ID_INT,
        (Xil_ExceptionHandler)XIntc_InterruptHandler, &intc);
    Xil_ExceptionEnable();
#endif
    return status;
}

static void handle_line(const char *line)
{
    uint16_t value = 0;
    u8 pwm_percent = 0U;
    // (추가) 08장 ADC READ / ADC PWM 명령은 기존 명령과 독립적으로 파싱.
    AdcCommand adc_cmd = adc_parse_command(line);
    if (adc_cmd != ADC_NOT_COMMAND) {
        if (adc_cmd == ADC_READ) {
            adc_print_reading();
        } else if (adc_cmd == ADC_PWM) {

static void poll_uart(void)
{
    static char line[LINE_CAP];
    static u32 used = 0;
    static int invalid = 0;
    static int overflow = 0;
    u8 ch;
    if (XUartLite_IsReceiveEmpty(UART_BASEADDR)) return;
    ch = XUartLite_RecvByte(UART_BASEADDR);
    if (ch == '\r' || ch == '\n') {
        line[used] = '\0';
        if (overflow) uart_puts("ERROR: line too long\r\n");
        else if (invalid) uart_puts("ERROR: invalid ASCII\r\n");
        else if (used) handle_line(line);
        used = 0; overflow = 0; invalid = 0;
        return; // CRLF second terminator is ignored.
    }
    if (ch < 0x20U || ch > 0x7EU) { invalid = 1; return; }
    if (used < LINE_CAP-1U) line[used++] = (char)ch;
    else overflow = 1;
}

module motor_controller #(
    parameter integer DEADTIME_CYCLES = 10000000, // 100 ms at 100MHz
    parameter integer FRESH_CYCLES    = 25000000  // 250 ms at 100MHz
)(
    input wire clk, input wire resetn,
    input wire [1:0] request_i, // 0 STOP, 1 FWD, 2 REV
    input wire stop_clear_i,   // One-clock pulse on MOTOR STOP write

    input wire sonic_valid_i, input wire sonic_unsafe_i,
    output wire motor_in1_o, output wire motor_in2_o, output wire motor_ena_o,
    output wire [1:0] active_dir_o,

    output reg trip_latched_o, output wire deadtime_o
);

    reg [1:0] drive_dir;
    // Always gate the physical outputs, even before the next clock edge.
    assign fresh_o = (fresh_count != 32'd0);
    assign sensor_safe_o = sonic_auto_i && fresh_o &&
                           sonic_valid_i && !sonic_unsafe_i;

                          !stop_clear_i && (request_i != 2'd0);

    assign motor_in2_o = output_allowed && (drive_dir == 2'd2);

    assign active_dir_o = motor_ena_o ? drive_dir : 2'd0;
    assign deadtime_o = (dead_count != 32'd0);

    always @(posedge clk) begin
        if (!resetn) begin
            fresh_count <= 32'd0;
            dead_count <= 32'd0;
            drive_dir <= 2'd0;
            trip_latched_o <= 1'b0;
        end else begin
            // Stale/invalid/off -> stop. Periodic successful DONE re-arms freshness.
            if (!sonic_auto_i || !sonic_valid_i || sonic_unsafe_i)
                fresh_count <= 32'd0;
            else if (sonic_done_i)
                fresh_count <= FRESH_CYCLES;
            else if (fresh_count != 32'd0)
                fresh_count <= fresh_count - 32'd1;

            if (stop_clear_i)
                trip_latched_o <= 1'b0;
            else if (request_i != 2'd0 && !sensor_safe_o)
                trip_latched_o <= 1'b1;

            if (!sensor_safe_o || trip_latched_o || stop_clear_i ||
                request_i == 2'd0 || request_i == 2'd3) begin
                drive_dir <= 2'd0;
                // (변경) STOP after motion must not bypass reversal dead-time.

                    dead_count <= dead_count - 32'd1;

                // Disable bridge, then enforce a direction-change dead time.
                drive_dir <= 2'd0;
                dead_count <= DEADTIME_CYCLES;

                if (dead_count != 32'd0)
                    dead_count <= dead_count - 32'd1;
                else
                    drive_dir <= request_i;
            end
        end
    end

`timescale 1ns/1ps
// P2-06 HC-SR04 reusable timing core; no board pins or AXI dependency.
// (추가) 2FF Echo synchronizer, microsecond tick, bounded waits, cooldown.
// 100 MHz AXI clock -> 100 clocks / 1 us; 12 us TRIG, >=60 ms cooldown.
// Result is the last COMPLETED reading. valid_o and fault flags persist.
module hc_sr04_core #(
    parameter integer CLK_HZ = 100000000,
    parameter integer TRIG_US = 12,
    parameter integer TIMEOUT_US = 38000,
    parameter integer COOLDOWN_US = 60000
)(
    input wire clk, input wire resetn,
    input wire echo_i, input wire start_i,
    output reg trig_o, output wire ready_o, output wire busy_o,
    output reg done_pulse_o,
    output reg [15:0] distance_cm_o,
    output reg [31:0] echo_us_o,
    output reg valid_o, output reg timeout_o, output reg out_range_o
);
    localparam integer TICK_CYCLES = CLK_HZ / 1000000;
    localparam [2:0] S_IDLE=3'd0, S_TRIG=3'd1,
                     S_WAIT=3'd2, S_ECHO=3'd3, S_COOL=3'd4;
    reg [2:0] state;
    reg [31:0] div_count, us_count, high_us;
    reg echo_meta, echo_sync, echo_prev;
    wire tick_us = (div_count == TICK_CYCLES-1);
    wire echo_rise = echo_sync && !echo_prev;
    reg echo_rise_pending; // Catch sub-us rising edges until the next us tick.
    assign ready_o = (state == S_IDLE);
    assign busy_o = !ready_o;

    // (추가) FPGA input can be asynchronous. Metastability cannot be
    // eliminated absolutely; this 2FF path lowers the risk.
    always @(posedge clk) begin
        if (!resetn) begin
            echo_meta <= 1'b0;
            echo_sync <= 1'b0;
            echo_prev <= 1'b0;
        end else begin
            echo_meta <= echo_i;
            echo_sync <= echo_meta;
            echo_prev <= echo_sync;
        end
    end

    always @(posedge clk) begin
        if (!resetn || state != S_WAIT) echo_rise_pending <= 1'b0;
        else if (echo_rise) echo_rise_pending <= 1'b1;
        else if (tick_us) echo_rise_pending <= 1'b0;
    end

    always @(posedge clk) begin
        if (!resetn) begin
            div_count <= 32'd0;
            state <= S_IDLE;
            us_count <= 32'd0;
            high_us <= 32'd0;
            trig_o <= 1'b0;
            done_pulse_o <= 1'b0;
            distance_cm_o <= 16'd0;
            echo_us_o <= 32'd0;
            valid_o <= 1'b0;
            timeout_o <= 1'b0;
            out_range_o <= 1'b0;
        end else begin
            done_pulse_o <= 1'b0;
            if (tick_us) div_count <= 32'd0;
            else div_count <= div_count + 32'd1;

            case (state)
            S_IDLE: begin
                if (start_i) begin
                    trig_o <= 1'b1;
                    us_count <= 32'd0;
                    high_us <= 32'd0;
                    state <= S_TRIG;
                end
            end
            S_TRIG: if (tick_us) begin
                if (us_count + 32'd1 >= TRIG_US) begin
                    trig_o <= 1'b0;
                    us_count <= 32'd0;
                    state <= S_WAIT;
                end else us_count <= us_count + 32'd1;
            end
            S_WAIT: if (tick_us) begin
                if (echo_rise_pending || echo_rise) begin
                    // Edge latched at 100MHz to avoid missing it on us tick.
                    high_us <= 32'd1;
                    us_count <= 32'd0;
                    state <= S_ECHO;
                end else if (us_count + 32'd1 >= TIMEOUT_US) begin
                    valid_o <= 1'b0;
                    timeout_o <= 1'b1;
                    out_range_o <= 1'b0;
                    distance_cm_o <= 16'd0;
                    echo_us_o <= 32'd0;
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else us_count <= us_count + 32'd1;
            end
            S_ECHO: if (tick_us) begin
                if (!echo_sync) begin
                    echo_us_o <= high_us;
                    // D(cm) ~= EchoHigh(us)/58 at nominal room temperature.
                    if (high_us >= 32'd116 && high_us <= 32'd23200) begin
                        distance_cm_o <= (high_us + 32'd29) / 32'd58;
                        valid_o <= 1'b1;
                        timeout_o <= 1'b0;
                        out_range_o <= 1'b0;
                    end else begin
                        distance_cm_o <= 16'd0;
                        valid_o <= 1'b0;
                        timeout_o <= 1'b0;
                        out_range_o <= 1'b1;
                    end
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else if (high_us >= TIMEOUT_US) begin
                    echo_us_o <= high_us;
                    distance_cm_o <= 16'd0;
                    valid_o <= 1'b0;
                    timeout_o <= 1'b1;
                    out_range_o <= 1'b0;
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else high_us <= high_us + 32'd1;
            end
            S_COOL: if (tick_us) begin
                if (us_count + 32'd1 >= COOLDOWN_US) begin
                    us_count <= 32'd0;
                    state <= S_IDLE;
                end else us_count <= us_count + 32'd1;
            end
            default: state <= S_IDLE;
            endcase
        end
    end
endmodule

`timescale 1ns/1ps
// P2-04: 보드와 무관한 단일 채널 범용 PWM 코어.
// (추가) period와 duty는 클록 횟수 단위. 0%/100%는 지속 LOW/HIGH.
// (추가) 설정은 주기 마지막 경계에서만 활성값으로 반영.
module pwm_core #(
    parameter [31:0] RESET_PERIOD = 32'd100000
)(
    input  wire        clk,
    input  wire        resetn,
    input  wire        cfg_apply,
    input  wire [31:0] cfg_period,
    input  wire [31:0] cfg_duty,
    input  wire        cfg_enable,
    input  wire        cfg_polarity,
    output wire        pwm_out,
    output reg         pending,
    output reg [31:0] active_period,
    output reg [31:0] active_duty,
    output reg         active_enable,
    output reg         active_polarity
);
    reg [31:0] counter;
    reg [31:0] shadow_period;
    reg [31:0] shadow_duty;
    reg        shadow_enable;
    reg        shadow_polarity;

    wire boundary = (counter >= active_period - 32'd1);
    wire duty_high = (counter < active_duty);

    // (주의) enable=0이면 polarity와 관계없이 LOW로 차단.
    // (설명) 기본 polarity에서 active_duty=0이면 항상 LOW,
    //         active_duty=active_period이면 항상 HIGH.
    assign pwm_out = resetn && active_enable &&
                     (active_polarity ? !duty_high : duty_high);

    always @(posedge clk) begin
        if (!resetn) begin
            counter         <= 32'd0;
            active_period   <= RESET_PERIOD;
            active_duty     <= 32'd0;
            active_enable   <= 1'b0;
            active_polarity <= 1'b0;
            shadow_period   <= RESET_PERIOD;
            shadow_duty     <= 32'd0;
            shadow_enable   <= 1'b0;
            shadow_polarity <= 1'b0;
            pending         <= 1'b0;
        end else begin
            if (boundary) begin
                counter <= 32'd0;
                if (pending) begin
                    active_period   <= shadow_period;
                    active_duty     <= shadow_duty;
                    active_enable   <= shadow_enable;
                    active_polarity <= shadow_polarity;
                    pending         <= 1'b0;
                end
            end else begin
                counter <= counter + 32'd1;
            end
            // (추가) 경계와 APPLY가 겹치면 새 값은 다음 경계에 적용.
            // 여러 APPLY가 겹치면 최종 요청값을 우선하는 단일 Shadow 구조.
            if (cfg_apply) begin
                shadow_period   <= cfg_period;
                shadow_duty     <= cfg_duty;
                shadow_enable   <= cfg_enable;
                shadow_polarity <= cfg_polarity;
                pending         <= 1'b1;
            end
        end
    end
endmodule
