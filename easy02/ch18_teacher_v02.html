00과제 개요

HC-SR501 PIR(Passive Infrared, 피아이알, 수동형 적외선) 센서 모듈은 주변 적외선 변화로 움직임을 감지하면 디지털 출력 OUT을 HIGH로 만든다. FPGA는 이 신호를 단순 GPIO 상태로 읽는 것에서 나아가 비동기 입력 동기화 → 상승 이벤트 검출 → CPU 인터럽트 발생 → ACK 해제로 확장한다.

기본 과제는 센서 초기 안정화, 입력 안정화 필터, 이벤트 횟수 및 인터럽트 Pending 등록이다. 응용 단계에서는 신규 AXI4-Lite 슬레이브 IP를 통해 ARM, IRQ_ENABLE, STATUS, EVENT_COUNT를 제어한다.

실제 HC-SR501 보드·MicroBlaze V IRQ 핸들러·Vivado 합성·타이밍·Vitis ELF 실행은 보류 상태다. 이 장의 PASS는 Icarus Verilog RTL 및 AXI4-Lite 단위 시뮬레이션으로 한정한다.

01전체 계층 구조

각 RTL 명칭은 실제 모듈과 일치한다. 이전 단계의 GPIO/AXI IP를 덮어쓰지 않고, 독립 PIR IRQ IP를 추가한다. 사용자 Top은 보드 핀·Wire·IP 연결만 담당한다.

센서 출력이 10초 동안 계속 HIGH라고 해도 이 설계에서는 한 번의 상승 전환을 1회로 기록한다. 반대로 센서가 LOW로 돌아왔다가 다시 HIGH가 되면 새로운 이벤트로 등록할 수 있다. event_count는 사람 수를 측정하는 값이 아니다.

ACK를 쓰는 바로 그 클록에 새로운 PIR 상승 에지가 들어오면 새로운 이벤트가 우선하며 Pending이 유지된다. CPU가 IRQ를 껐다 켜도 Pending은 그대로 남아 있을 수 있다.

`timescale 1ns/1ps
// P2-18 HC-SR501 PIR event controller (board-independent).
// pir_in_async -> 2-FF synchronizer -> stable filter -> rising event.
// Warm-up masks initial module transients. Pending is sticky until ACK;
// if ACK and new event coincide, NEW EVENT WINS (no lost interrupt).
// IRQ is level-sensitive: irq_enable & event_pending.
// This RTL does NOT implement a physical AXI-Lite bus / CPU interrupt controller.
module pir_irq_controller #(
    parameter [63:0] WARMUP_CYCLES = 64'd6000000000, // 60 sec @100 MHz
    parameter integer FILTER_CYCLES = 3
)(
    input wire clk,
    input wire resetn,
    input wire pir_in_async,
    input wire arm,
    input wire irq_enable,
    input wire irq_ack_pulse,
    input wire count_clear_pulse,
    output wire irq,
    output reg warmup_ready,
    output reg pir_level,
    output reg motion_pulse,
    output reg event_pending,
    output reg [31:0] event_count
);
    // Wide enough for 60s * 100MHz, no 32-bit overflow.
    reg [63:0] warmup_count;
    reg [31:0] filter_count;
    reg pir_meta, pir_sync;
    wire change_pending = (pir_sync != pir_level);
    wire qualified_rise = warmup_ready && arm && !pir_level &&
        pir_sync && change_pending && (filter_count >= FILTER_CYCLES-1);
    assign irq = irq_enable && event_pending;

    always @(posedge clk) begin
        if(!resetn) begin
            pir_meta<=0; pir_sync<=0; pir_level<=0;
            filter_count<=0; warmup_count<=0; warmup_ready<=0;
            motion_pulse<=0; event_pending<=0; event_count<=0;
        end else begin
            // (추가) Async PIR input synchronizer.
            pir_meta<=pir_in_async;
            pir_sync<=pir_meta;
            if(!warmup_ready) begin
                if(warmup_count >= WARMUP_CYCLES-1)
                    warmup_ready<=1;
                else warmup_count<=warmup_count+1;
            end

            // (추가) Reject glitches shorter than FILTER_CYCLES.
            if(!change_pending) filter_count<=0;
            else if(filter_count >= FILTER_CYCLES-1) begin
                pir_level<=pir_sync;
                filter_count<=0;
            end else filter_count<=filter_count+1;

            // Sticky event; simultaneous ACK cannot erase a new edge.
            motion_pulse<=qualified_rise;
            if(qualified_rise)
                event_pending<=1;
            else if(irq_ack_pulse)
                event_pending<=0;

            // Saturating number of qualified rising edges. This is NOT
            // a direct count of humans or motion events behind one long
            // HC-SR501 HIGH interval.
            if(count_clear_pulse) event_count<=qualified_rise ? 1 : 0;
            else if(qualified_rise && event_count!=32'hffffffff)
                event_count<=event_count+1;
        end
    end
endmodule

// Register offsets:
// 0x00 CTRL R/W: bit0 ARM, bit1 IRQ_ENABLE
// 0x04 CMD  W: bit0 ACK pending, bit1 CLEAR event_count (W1P)

// 0x0C EVENT_COUNT R
// 0x10 ID R=0x00180001

// GPIO input asynchronous. CPU IRQ is LEVEL (not a one-clock pulse).

    parameter integer FILTER_CYCLES=3
)(

);

    reg [31:0] saved_data;
    reg [3:0] saved_strb;

        .WARMUP_CYCLES(WARMUP_CYCLES),.FILTER_CYCLES(FILTER_CYCLES)
    ) u_core(

`timescale 1ns/1ps
module tb_pir_irq_controller;
    reg clk=0; always #5 clk=~clk;
    reg resetn=0, pir=0, arm=1, enable=1, ack=0, clear=0;
    wire irq,ready,level,pulse,pending;
    wire [31:0] count;
    integer pulses=0;
    pir_irq_controller #(
        .WARMUP_CYCLES(64'd20), .FILTER_CYCLES(3)
    ) dut (
        .clk(clk),.resetn(resetn),.pir_in_async(pir),
        .arm(arm),.irq_enable(enable),
        .irq_ack_pulse(ack),.count_clear_pulse(clear),
        .irq(irq),.warmup_ready(ready),.pir_level(level),
        .motion_pulse(pulse),.event_pending(pending),.event_count(count)
    );
    always @(posedge clk) if (pulse) pulses=pulses+1;
    task advance;
        input integer cycles;
        integer k;
        begin for(k=0;k<cycles;k=k+1) @(negedge clk); end
    endtask
    task set_pir;
        input value;
        begin @(negedge clk);pir=value; end
    endtask
    task do_ack;
        begin @(negedge clk);ack=1;@(negedge clk);ack=0;end
    endtask
    initial begin
        advance(4);resetn=1;
        set_pir(1);advance(9);
        if(ready || count!=0 || pending || irq)
            $fatal(1,"warmup early event");
        advance(20);
        if(!ready || !level || pending || count!=0)
            $fatal(1,"warmup high should be baseline, not event");
        $display("P2_18_PIR_WARMUP_MASK_PASS");

        set_pir(0);advance(9);
        if(level!==0) $fatal(1,"falling edge not synchronized");
        // 1-cycle glitch is shorter than FILTER_CYCLES, ignore.
        set_pir(1);advance(1);set_pir(0);advance(9);
        if(level || count || pending) $fatal(1,"glitch counted");
        $display("P2_18_PIR_GLITCH_FILTER_PASS");

        set_pir(1);advance(11);
        if(!level || !pending || !irq || count!=1 || pulses!=1)
            $fatal(1,"rising event count=%0d pulses=%0d",count,pulses);
        advance(30);
        if(count!=1 || pulses!=1) $fatal(1,"stuck HIGH duplicate event");
        $display("P2_18_PIR_RISE_ONLY_STICKY_IRQ_PASS");

        // Mask IRQ without clearing pending; re-enable reveals pending.
        @(negedge clk);enable=0;
        #1;
        if(irq || !pending) $fatal(1,"enable mask failure");
        @(negedge clk);enable=1;
        #1;
        if(!irq) $fatal(1,"unmask pending failure");
        do_ack();advance(2);
        if(irq || pending || count!=1)
            $fatal(1,"ack failed or counter lost");
        $display("P2_18_PIR_IRQ_ENABLE_ACK_PASS");

        // Returning LOW and then HIGH creates another event.
        set_pir(0);advance(9);
        set_pir(1);advance(10);
        if(count!=2 || !irq || pulses!=2)
            $fatal(1,"second motion not counted");
        do_ack();
        $display("P2_18_PIR_RETRIGGER_PASS");

        // Disable ARM, next high does not create event;
        // enabling ARM while sensor stays high also does not.
        set_pir(0);advance(10);
        @(negedge clk);arm=0;
        set_pir(1);advance(10);
        if(count!=2 || pending || irq)
            $fatal(1,"arm gate failed");
        @(negedge clk);arm=1;advance(7);
        if(count!=2 || pending) $fatal(1,"arm caused fake rise");
        $display("P2_18_PIR_ARM_GATE_PASS");

        set_pir(0);advance(10);
        // ACK coincides with new event: event MUST win.
        set_pir(1);
        wait(dut.qualified_rise);
        @(negedge clk);ack=1;
        @(negedge clk);ack=0;
        advance(4);
        if(count!=3 || !pending || !irq || pulses!=3)
            $fatal(1,"simultaneous ACK erased event");
        $display("P2_18_PIR_ACK_EVENT_PRIORITY_PASS");

        // Clear counter, but not the independent pending bit.
        @(negedge clk);clear=1;
        @(negedge clk);clear=0;
        advance(2);
        if(count!=0 || !pending) $fatal(1,"counter clear affected pending");
        do_ack();
        if(pending || irq || count!=0)
            $fatal(1,"final ACK failed");
        $display("P2_18_PIR_COUNT_CLEAR_PASS");
        $display("P2_18_PIR_IRQ_CONTROLLER_RTL PASS");
        $finish;
    end
    initial begin #200000; $fatal(1,"TIMEOUT");end
endmodule
