00과제 개요

BME280(비엠이280, 기압·온도·습도 센서)와 기존 DHT22, DS18B20, DS3231 RTC, PIR 센서의 데이터를 센서별 독립 인터페이스 → 공통 제어 정책 → 표시·경보로 통합한다. 핵심은 센서 종류가 늘어나도 기존 Controller와 Top의 기능 로직을 변경하지 않는 것이다.

BME280 Chip-ID·Raw ADC 레지스터, 센서 선택 우선순위, 경보 정책과 표시 페이지를 C/RTL 자가검증으로 확인한다.

검증 경계: BME280 Raw ADC로부터 실제 ℃·%RH·Pa로 보정하는 Bosch Calibration/Compensation 코드는 미구현이다. 테스트벤치의 BME280 입력은 상류에서 이미 보정했다고 가정한 가상값이다. 실제 I²C 버스 결합, AXI IP, Vivado·Vitis·실물 검증은 아직 수행하지 않았다.

01전체 계층형 구조도

기존 14~18장 RTL은 수정하지 않는다. 각 센서에서 검증된 값이 들어오면 정책 모듈이 저장·표시·경보를 결정한다. 상류 센서 실제 출력과 이 모듈을 연결하는 Adapter(어댑터, 인터페이스 변환기), AXI 통합, I²C 공용 버스는 아직 실행 검증을 마치지 않았다.

온도 선택 순서는 BME280 → DHT22 → DS18B20, 습도 선택 순서는 BME280 → DHT22이다. 선택한 BME280 값이 물리 범위를 벗어나면 DHT22 또는 DS18B20으로 되돌아간다.

기존 센서의 단위가 서로 달라도, 정책 계층에서는 0.1℃·0.1%RH 형태의 일관된 값을 처리하도록 정의한다. 이 단계에서 데이터 전환 우선순위가 바뀔 수는 있지만 센서별 최신 시각이 일치한다는 보장은 아직 없다.

중요: ID가 0x58인 BMP280(비엠피280)은 기압·온도 센서이며 BME280과 다르다. BME280의 실제 물리량은 보정 계수를 이용해 Bosch 공식 알고리즘을 적용해야 한다. Raw ADC 값을 온도나 기압으로 표시하면 기술적으로 잘못된 결과가 된다.

공식 근거: Bosch BME280 데이터시트 · Bosch 공식 BME280 SensorAPI.

05전체 RTL·C·자가검증 테스트벤치

정책 컨트롤러는 기존 센서를 직접 변경하지 않고 이미 검증된 데이터만 수신한다. Bosch의 BME280 레지스터 구조를 확인하는 C 드라이버는 추후 I²C Read 구현과 연결할 수 있는 Callback(콜백, 외부 통신 함수를 전달하는 인터페이스) 형태다.

rtl/environment_policy.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// P2-19 Reusable integrated environment policy controller.
// Inputs are ALREADY DECODED sample values: not a physical I2C driver.
// DHT22: 0.1%RH + signed 0.1C; DS18B20: signed 1/16C.
// BME280: OPTIONAL upstream compensated 0.01C, 0.001%RH and Pa.
// RTC: ALREADY DECODED 24h hour/minute. PIR input is sticky event pending.
// Priority: PIR > TEMP HIGH > HUM HIGH > OK. No hardware actuator output.
// Display round-robin pages: temperature, humidity, pressure, RTC, PIR.
// Page selection is a presentation policy, not an LCD byte-level protocol.
module environment_policy #(
    parameter signed [15:0] TEMP_ALARM_TENTHS = 16'sd300,
    parameter [15:0] HUM_ALARM_TENTHS = 16'd800
)(
    input wire clk, input wire resetn,
    input wire dht_valid, input wire signed [15:0] dht_temp_tenths,
    input wire [15:0] dht_hum_tenths,
    input wire ds_valid, input wire signed [15:0] ds_temp_raw,
    input wire bme_valid, input wire signed [31:0] bme_temp_centi,
    input wire [31:0] bme_hum_milli, input wire [31:0] bme_pressure_pa,
    input wire rtc_valid, input wire [4:0] rtc_hour, input wire [5:0] rtc_minute,
    input wire pir_event_pending,
    input wire page_next_pulse,
    output reg [2:0] display_page,
    output reg display_valid,
    output reg signed [31:0] display_value,
    output reg [2:0] display_source, // 1 DHT, 2 DS, 3 BME, 4 RTC, 5 PIR
    output reg [2:0] alarm_reason, // 0 none, 1 PIR, 2 high-temp, 3 high-humidity
    output wire alarm_active,
    output reg [31:0] accepted_updates
);
    reg dht_seen,ds_seen,bme_seen,rtc_seen;
    reg signed [15:0] dht_t;
    reg [15:0] dht_h;
    reg signed [15:0] ds_t;
    reg signed [31:0] bme_t;
    reg [31:0] bme_h,bme_p;
    reg [4:0] hour_reg;
    reg [5:0] min_reg;

    wire bme_temp_sane = (bme_t >= -32'sd4000 &&
                           bme_t <= 32'sd8500);
    wire bme_hum_sane = (bme_h <= 32'd100000);
    wire bme_pressure_sane = (bme_p >= 32'd30000 &&
                              bme_p <= 32'd110000);
    wire bme_usable = bme_seen && bme_temp_sane && bme_hum_sane &&
                      bme_pressure_sane;
    wire dht_usable = dht_seen && dht_h<=16'd1000 &&
                      dht_t>=-16'sd400 && dht_t<=16'sd800;
    wire ds_usable = ds_seen && ds_t>=-16'sd880 &&
                     ds_t<=16'sd2000;
    wire rtc_usable = rtc_seen && hour_reg<=5'd23 &&
                      min_reg<=6'd59;
    // Select one coherent temperature source; bme is preferred if valid.
    reg signed [31:0] temp_selected;
    reg [31:0] hum_selected;
    reg temp_available,hum_available;
    reg [2:0] temp_src,hum_src;
    always @(*) begin
        temp_selected=0;hum_selected=0;
        temp_available=0;hum_available=0;
        temp_src=0;hum_src=0;
        if(bme_usable) begin
            temp_selected=bme_t / 32'sd10; // centi-C to tenth-C
            hum_selected=bme_h / 32'd100;  // milli-% to tenth-%
            temp_available=1;hum_available=1;
            temp_src=3;hum_src=3;
        end else begin
            if(dht_usable) begin
                temp_selected=dht_t;hum_selected=dht_h;
                temp_available=1;hum_available=1;
                temp_src=1;hum_src=1;
            end else if(ds_usable) begin
                // signed signed raw, convert 1/16C into 1/10C;
                // integer truncates towards zero, acceptable display approx.
                temp_selected=($signed(ds_t)*32'sd10)/32'sd16;
                temp_available=1;temp_src=2;
            end
        end
    end
    assign alarm_active=(alarm_reason!=0);
    always @(*) begin
        alarm_reason=0;
        if(pir_event_pending) alarm_reason=1;
        else if(temp_available &&
                temp_selected >= TEMP_ALARM_TENTHS)
            alarm_reason=2;
        else if(hum_available &&
                hum_selected >= HUM_ALARM_TENTHS)
            alarm_reason=3;
    end
    always @(*) begin
        display_valid=0;display_value=0;display_source=0;
        case(display_page)
        0:if(temp_available)begin
            display_valid=1;display_value=temp_selected;
            display_source=temp_src;
        end
        1:if(hum_available)begin
            display_valid=1;display_value=$signed(hum_selected);
            display_source=hum_src;
        end
        2:if(bme_usable)begin
            display_valid=1;display_value=$signed(bme_p);
            display_source=3;
        end
        3:if(rtc_usable)begin
            display_valid=1;
            display_value=hour_reg*32'd100+min_reg;
            display_source=4;
        end
        4:begin
            display_valid=1;
            display_value=pir_event_pending ? 32'sd1 : 32'sd0;
            display_source=5;
        end
        default:begin display_valid=0;end
        endcase
    end
    always @(posedge clk) begin
        if(!resetn)begin
            dht_seen<=0;ds_seen<=0;bme_seen<=0;rtc_seen<=0;
            dht_t<=0;dht_h<=0;ds_t<=0;
            bme_t<=0;bme_h<=0;bme_p<=0;hour_reg<=0;min_reg<=0;
            display_page<=0;accepted_updates<=0;
        end else begin
            if(page_next_pulse)
                display_page <= (display_page==4) ? 0 : display_page+1'b1;
            if(dht_valid) begin
                dht_seen<=1;dht_t<=dht_temp_tenths;dht_h<=dht_hum_tenths;
            end
            if(ds_valid) begin ds_seen<=1;ds_t<=ds_temp_raw;end
            if(bme_valid) begin
                bme_seen<=1;bme_t<=bme_temp_centi;
                bme_h<=bme_hum_milli;bme_p<=bme_pressure_pa;
            end
            if(rtc_valid) begin
                rtc_seen<=1;hour_reg<=rtc_hour;min_reg<=rtc_minute;
            end
            // Count ACCEPTED DATA update cycles, not physical sensors.
            if(dht_valid || ds_valid || bme_valid || rtc_valid)
                if(accepted_updates!=32'hffffffff)
                    accepted_updates<=accepted_updates+1;
        end
    end
endmodule

#include "bme280_probe.h"
// Bosch Sensortec BME280: 7bit addr 0x76/0x77; ID register D0 -> 0x60.
// Data 0xF7..0xFE: pressure MSB/LSB/XLSB, temp, then humidity.
// Compensation needs device-specific calibration registers and Bosch's
// compensation algorithms. This helper exposes ONLY raw ADC words.
static int address_ok(uint8_t addr7)
{
    return addr7==0x76u || addr7==0x77u;
}
bme280_probe_status_t bme280_probe_id(bme280_read_fn read_reg,
    void *ctx,uint8_t addr7)
{
    uint8_t id=0;
    if(!read_reg || !address_ok(addr7)) return BME280_PROBE_BAD_ARG;
    if(read_reg(ctx,addr7,0xd0u,&id,1)!=0) return BME280_PROBE_BUS;
    return id==0x60u ? BME280_PROBE_OK : BME280_PROBE_NOT_BME280;
}
bme280_probe_status_t bme280_read_raw8(bme280_read_fn read_reg,
    void *ctx,uint8_t addr7,bme280_raw_t *raw)
{
    uint8_t bytes[8];
    bme280_raw_t decoded;
    if(!read_reg || !raw || !address_ok(addr7))
        return BME280_PROBE_BAD_ARG;
    if(read_reg(ctx,addr7,0xf7u,bytes,sizeof(bytes))!=0)
        return BME280_PROBE_BUS;
    decoded.adc_pressure_20 = ((uint32_t)bytes[0]<<12) |
          ((uint32_t)bytes[1]<<4) | (uint32_t)(bytes[2]>>4);
    decoded.adc_temperature_20 = ((uint32_t)bytes[3]<<12) |
          ((uint32_t)bytes[4]<<4) | (uint32_t)(bytes[5]>>4);
    decoded.adc_humidity_16 = (uint16_t)(((uint16_t)bytes[6]<<8) |
                                            (uint16_t)bytes[7]);
    *raw=decoded;
    return BME280_PROBE_OK;
}

        begin @(negedge clk);end

        input v;
        input signed [31:0] n;

                    page,valid,value,source,p,v,n,src);
        end

        check_display(0,0,0,0);

#include "../sw/bme280_probe.h"
#include <assert.h>
#include <string.h>
#include <stdio.h>
typedef struct {
    uint8_t regs[256];
    uint8_t captured_addr, captured_reg;
    size_t captured_len;
    int fail;
} bus_t;
static int mock_read(void *ctx,uint8_t a,uint8_t reg,uint8_t *d,size_t n)
{
    bus_t *bus=(bus_t*)ctx;
    bus->captured_addr=a;bus->captured_reg=reg;bus->captured_len=n;
    if(bus->fail || (size_t)reg+n>256u)return -1;
    memcpy(d,bus->regs+reg,n);
    return 0;
}
int main(void)
{
    bus_t b={0};
    bme280_raw_t raw={0};
    b.regs[0xd0]=0x60;
    assert(bme280_probe_id(mock_read,&b,0x76)==BME280_PROBE_OK);
    assert(b.captured_addr==0x76 && b.captured_reg==0xd0 &&
           b.captured_len==1);
    b.regs[0xd0]=0x58; // BMP280 is pressure/temp only; not BME280.
    assert(bme280_probe_id(mock_read,&b,0x77)==BME280_PROBE_NOT_BME280);
    assert(bme280_probe_id(mock_read,&b,0x75)==BME280_PROBE_BAD_ARG);
    puts("P2_19_BME280_ID_BMP280_REJECT_PASS");
    // Raw pressure 0xABCDE, raw temperature 0x54321, humidity 0xA55A.
    const uint8_t data[8]={0xab,0xcd,0xe0,0x54,0x32,0x10,0xa5,0x5a};
    memcpy(b.regs+0xf7,data,8);
    assert(bme280_read_raw8(mock_read,&b,0x77,&raw)==BME280_PROBE_OK);
    assert(raw.adc_pressure_20==0xabcde &&
           raw.adc_temperature_20==0x54321 &&
           raw.adc_humidity_16==0xa55a);
    assert(b.captured_addr==0x77 && b.captured_reg==0xf7 &&
           b.captured_len==8);
    puts("P2_19_BME280_RAW8_DECODER_PASS");
    b.fail=1;
    assert(bme280_probe_id(mock_read,&b,0x76)==BME280_PROBE_BUS);
    assert(bme280_read_raw8(mock_read,&b,0x76,&raw)==BME280_PROBE_BUS);
    puts("P2_19_BME280_BUS_ERROR_PASS");
    puts("P2_19_BME280_HOST_C_PASS");
    return 0;
}
