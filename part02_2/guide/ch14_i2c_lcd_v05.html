00과제 개요

I²C(아이투씨, 두 선을 이용하는 직렬 버스)는 SDA(데이터)와 SCL(클록)을 이용한다. FPGA(재구성 가능한 디지털 회로 칩)의 I²C Master가 PCF8574(아이투씨 입출력 확장기)에 8비트 데이터를 보내면, PCF8574가 캐릭터 LCD의 RS·RW·E·D4~D7 핀을 대신 구동한다.

한 단계 더 나아가, HD44780 계열 LCD의 4비트 초기화 규격을 따른 뒤 첫 번째 줄에 ASCII 문자열 HELLO를 쓰는 과제다. LCD1602의 2줄과 LCD2004의 4줄 출력은 공통으로 첫 줄 DDRAM 주소 0x00을 이용한다.

이번 장의 검증 범위는 RTL 시뮬레이션이다. PCF8574 가짜 ACK 응답으로 I²C 주소·데이터 및 LCD 명령 72회를 확인한다. 실제 보드의 합성·타이밍·Vitis/AXI 전체 연결·LCD 문자 출력은 현 단계에서 보류한다.

01전체 구조

세 계층은 LCD 명령 처리 → I²C 전송 → 물리 오픈드레인 핀으로 분리한다. 기존 Top에 LCD 상태 머신이나 프로토콜 비교·산술을 넣지 않는다.

02I²C 기본 프로토콜

신호뜻FPGA 기준
SCLSerial Clock(직렬 클록)LOW 구동 또는 High-Z 해제
SDASerial Data(직렬 데이터)LOW 구동 또는 High-Z 해제, ACK 입력 읽기
STARTSCL HIGH 상태에서 SDA가 HIGH→LOW슬레이브 시작 통보
ACK9번째 SCL 구간에서 슬레이브가 SDA LOW수신 성공
NACK9번째 SCL 구간에서 SDA HIGH주소 미응답/쓰기 실패
STOPSCL HIGH 상태에서 SDA가 LOW→HIGH트랜잭션 종료

한 번의 명령은 START → 주소+W → ACK → DATA → ACK → STOP으로 구성한다. 기본 7비트 슬레이브 주소는 0x27이며 실제 모듈에 따라 0x3F 등으로 바뀔 수 있다.

100MHz 시스템 클록·CLK_DIV=400 기준 설계된 SCL은 기본 바이트 전송에서 약 83kHz이며, PCF8574의 최대 100kHz Standard-mode 규격에 맞춘 보수적인 예제다. Clock Stretching(슬레이브의 클록 연장)과 Arbitration(다중 마스터 중재)은 지원하지 않는다.

이 배치는 흔한 PCF8574 LCD 백팩을 가정한 것이며 모든 제조사의 모듈에 동일하게 적용된다고 단정할 수 없다. RS가 0인 명령과 RS가 1인 문자 출력을 구분한다. LCD의 1개 4비트 니블을 실제로 출력할 때는 PCF8574에 E=0 → E=1 → E=0의 세 바이트를 연속 전송한다.

일부 LCD 모듈은 5V로 구동하면서 SDA/SCL을 5V에 풀업한다. Basys3는 3.3V I/O 기준이므로 **5V 풀업된 SDA/SCL을 직접 연결하면 안 된다.** LCD 모듈·백팩의 회로 및 입력 로직 임계값을 확인하고 필요하면 양방향 I²C 레벨 시프터를 사용한다.

첫 번째 3 니블 이후 5ms, 다음 초기화 단계에는 200µs를 기본 적용한다. LCD Clear 뒤에는 2ms를 준다. 초기화 규칙은 Hitachi HD44780U 공식 4-bit 초기화 그림을 따른다.

초기 니블 4개는 각 3회 I²C 쓰기, 이어지는 완전한 바이트 10개는 각각 6회 쓰기이므로 전체 PCF8574 전송 횟수는 12 + 60 = 72회다.

rtl/i2c_write_byte_master.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// P2-14: I2C single-byte write (START + 7-bit address/W + DATA + STOP).
// Both *_drive_low outputs are open-drain controls: 1=pull low, 0=release.
// External 3.3V pullups required; clock stretching/arbitration unsupported.
// FPGA clk is the only clock domain. SCL is an output, not an FPGA clock.
module i2c_write_byte_master #(
    parameter integer CLK_DIV = 400
)(
    input wire clk, input wire resetn,
    input wire cmd_valid, output wire cmd_ready,
    input wire [6:0] cmd_addr, input wire [7:0] cmd_data,
    input wire sda_in,
    output reg scl_drive_low, output reg sda_drive_low,
    output reg done_pulse, output reg nack_pulse
);
    localparam [3:0] IDLE=0, START=1, BIT_LOW=2, BIT_UP=3,
        BIT_END=4, ACK_LOW=5, ACK_UP=6, ACK_END=7,
        STOP_LOW=8, STOP_UP=9, STOP_END=10, COMPLETE=11;
    reg [3:0] state;
    reg [7:0] tx_byte, data_latched;
    reg [2:0] bit_index;
    reg byte_index;
    reg nack_seen;
    reg [31:0] ticks;
    wire tick = (ticks == CLK_DIV-1);
    assign cmd_ready = (state == IDLE);
    always @(posedge clk) begin
        if(!resetn) begin
            state<=IDLE;tx_byte<=0;data_latched<=0;
            bit_index<=0;byte_index<=0;nack_seen<=0;ticks<=0;
            scl_drive_low<=0;sda_drive_low<=0;
            done_pulse<=0;nack_pulse<=0;
        end else begin
            done_pulse<=0;nack_pulse<=0;
            if(state==IDLE || tick) ticks<=0;
            else ticks<=ticks+1;
            case(state)
            IDLE: begin
                scl_drive_low<=0;sda_drive_low<=0;
                if(cmd_valid) begin
                    tx_byte<={cmd_addr,1'b0};
                    data_latched<=cmd_data;
                    bit_index<=3'd7;byte_index<=0;nack_seen<=0;
                    state<=START;
                end
            end
            START: if(tick) begin
                // SDA falls while SCL is high: START.
                sda_drive_low<=1;
                state<=BIT_LOW;
            end
            BIT_LOW: if(tick) begin
                scl_drive_low<=1;
                sda_drive_low<=~tx_byte[bit_index];
                state<=BIT_UP;
            end
            BIT_UP: if(tick) begin
                scl_drive_low<=0;state<=BIT_END;
            end
            BIT_END: if(tick) begin
                scl_drive_low<=1;
                if(bit_index==0) state<=ACK_LOW;
                else begin bit_index<=bit_index-1;state<=BIT_LOW;end
            end
            ACK_LOW: if(tick) begin
                sda_drive_low<=0;state<=ACK_UP;
            end
            ACK_UP: if(tick) begin
                scl_drive_low<=0;state<=ACK_END;
            end
            ACK_END: if(tick) begin
                if(sda_in) nack_seen<=1;
                scl_drive_low<=1;
                // (변경) Address NACK 이후에는 data byte를 보내지 않고
                // 즉시 STOP을 생성한다. Data NACK도 STOP으로 종료.
                if(sda_in) state<=STOP_LOW;
                else if(!byte_index) begin
                    byte_index<=1;tx_byte<=data_latched;
                    bit_index<=3'd7;state<=BIT_LOW;
                end else state<=STOP_LOW;
            end
            STOP_LOW: if(tick) begin
                sda_drive_low<=1;state<=STOP_UP;
            end
            STOP_UP: if(tick) begin
                scl_drive_low<=0;state<=STOP_END;
            end
            STOP_END: if(tick) begin
                // SDA rising with SCL HIGH: STOP.
                sda_drive_low<=0;state<=COMPLETE;
            end
            COMPLETE: if(tick) begin
                done_pulse<=1;nack_pulse<=nack_seen;state<=IDLE;
            end
            default: state<=IDLE;
            endcase
        end
    end
endmodule
rtl/lcd_pcf8574_hello.v — 전체 코드 펼치기
rtl/lcd_pcf8574_hello.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// P2-14 · PCF8574 backpack 4-bit LCD initialization + "HELLO".
// Typical backpack pin mapping: P7..P4=D7..D4, P3=BL, P2=E, P1=RW, P0=RS.
// Does not assume all PCF8574 backpacks use the same mapping.
// Sends 3 I2C writes per LCD nibble: E=0 -> E=1 -> E=0.
// First 4 boot nibbles: 3,3,3,2; then 0x28,0x0C,0x06,0x01,0x80,
// then ASCII 'H','E','L','L','O'. Single-line subset of 1602/2004.
// Each I2C write is START/address/data/STOP (slow but easy to verify).
module lcd_pcf8574_hello #(
    parameter [6:0] I2C_ADDR=7'h27,
    parameter [31:0] POWERUP_WAIT_CYCLES=32'd5000000,
    // HD44780-compatible power-on initialization spacing:
    // first 0x3 nibble -> >=4.1ms, further 0x3/0x2 -> >=100us.
    parameter [31:0] FIRST_INIT_WAIT_CYCLES=32'd500000,
    parameter [31:0] OTHER_INIT_WAIT_CYCLES=32'd20000,
    parameter [31:0] CLEAR_WAIT_CYCLES=32'd200000
)(
    input wire clk, input wire resetn,
    output wire cmd_valid, input wire cmd_ready,
    output wire [6:0] cmd_addr, output wire [7:0] cmd_data,
    input wire cmd_done, input wire cmd_nack,
    output reg ready_done, output reg error_latched
);
    localparam [2:0] POWERUP=0,SEND=1,WAIT_RSP=2,GUARD=3,
                     FINISH=4,FAULT=5;
    reg [2:0] state;
    reg [31:0] guard_counter;
    reg [3:0] step_index;   // 0..13
    reg nibble_select;     // 0 high nibble, 1 low nibble
    reg [1:0] enable_phase;// 0=LOW, 1=HIGH, 2=LOW
    reg [7:0] lcd_byte;
    reg [3:0] nibble;
    reg rs;
    always @(*) begin
        lcd_byte=8'h00;
        case (step_index)
        4: lcd_byte=8'h28;  // 4-bit mode, 2 rows, 5x8 glyph
        5: lcd_byte=8'h0C;  // display ON, cursor OFF
        6: lcd_byte=8'h06;  // entry increment
        7: lcd_byte=8'h01;  // clear display
        8: lcd_byte=8'h80;  // DDRAM line-1 address 0
        9: lcd_byte=8'h48;  // H
        10: lcd_byte=8'h45; // E
        11: lcd_byte=8'h4C; // L
        12: lcd_byte=8'h4C; // L
        13: lcd_byte=8'h4F; // O
        default: lcd_byte=8'h00;
        endcase
        rs=(step_index>=9);
        if(step_index<4) begin
            case(step_index)
            0,1,2: nibble=4'h3;
            default: nibble=4'h2;
            endcase
        end else
            nibble=nibble_select ? lcd_byte[3:0] : lcd_byte[7:4];
    end
    assign cmd_addr=I2C_ADDR;
    assign cmd_valid=(state==SEND);
    assign cmd_data={nibble,1'b1,(enable_phase==1),1'b0,rs};
    always @(posedge clk) begin
        if(!resetn) begin
            state<=POWERUP;guard_counter<=POWERUP_WAIT_CYCLES;
            step_index<=0;nibble_select<=0;enable_phase<=0;
            ready_done<=0;error_latched<=0;
        end else begin
            case(state)
            POWERUP: if(guard_counter==0) state<=SEND;
                     else guard_counter<=guard_counter-1;
            SEND: if(cmd_ready) state<=WAIT_RSP;
            WAIT_RSP: if(cmd_done) begin
                if(cmd_nack) begin
                    error_latched<=1; state<=FAULT;
                end else if(enable_phase!=2) begin
                    enable_phase<=enable_phase+1;
                    state<=SEND;
                end else begin
                    enable_phase<=0;
                    if(step_index>=4 && !nibble_select) begin
                        nibble_select<=1;state<=SEND;
                    end else begin
                        nibble_select<=0;
                        if(step_index==13) state<=FINISH;
                        else begin
                            step_index<=step_index+1;
                            // 0x01 clear display requires >=1.52ms settle time.
                            if(step_index==7) begin
                                guard_counter<=CLEAR_WAIT_CYCLES;
                                state<=GUARD;
                            end else if(step_index==0) begin
                                guard_counter<=FIRST_INIT_WAIT_CYCLES;
                                state<=GUARD;
                            end else if(step_index==1 || step_index==2 ||
                                        step_index==3) begin
                                guard_counter<=OTHER_INIT_WAIT_CYCLES;
                                state<=GUARD;
                            end else state<=SEND;
                        end
                    end
                end
            end
            GUARD: if(guard_counter==0) state<=SEND;
                   else guard_counter<=guard_counter-1;
            FINISH: ready_done<=1;
            FAULT: state<=FAULT;
            default: state<=FAULT;
            endcase
        end
    end
endmodule

module tb_lcd_i2c_hello_system;
    reg clk=0; always #5 clk=~clk;
    reg resetn=0;
    reg slave_ack_drive=0;
    wire scl_low,sda_low,ready_done,error_latched;
    wire scl = scl_low ? 1'b0 : 1'b1; // digital model of 3.3V pull-up
    wire sda = (sda_low || slave_ack_drive) ? 1'b0 : 1'b1;
    integer bit_pos=0, bytes_seen=0, writes_seen=0, transactions=0;
    integer nibble_index=0;
    reg [7:0] incoming=0;
    reg [7:0] address_byte=0;
    reg [7:0] pcf_byte=0;
    reg [7:0] expected [0:71];
    integer i;

    lcd_i2c_hello_system #(
        .CLK_DIV(3),.LCD_ADDR(7'h27),
        .POWERUP_WAIT_CYCLES(32'd4),
        .FIRST_INIT_WAIT_CYCLES(32'd4),
        .OTHER_INIT_WAIT_CYCLES(32'd4),.CLEAR_WAIT_CYCLES(32'd4)
    ) dut (
        .clk(clk),.resetn(resetn),.sda_in(sda),
        .scl_drive_low(scl_low),.sda_drive_low(sda_low),
        .ready_done(ready_done),.error_latched(error_latched)
    );

    // Fake PCF8574 ACK at each ninth clock, and capture both bytes.
    // This is not a real electrical level-shifter/pull-up test.
    always @(negedge sda) begin
        if(scl && resetn) begin
            bit_pos=0;
            incoming=0;
            bytes_seen=0;
        end
    end
    always @(negedge scl) begin
        if(bit_pos==8) slave_ack_drive=1;
        else slave_ack_drive=0;
    end
    always @(posedge scl) begin
        if(resetn && !ready_done) begin
            if(bit_pos<8) begin
                incoming={incoming[6:0],sda};
                bit_pos=bit_pos+1;
            end else begin
                if(sda!==0) $fatal(1,"ACK not low");
                if(bytes_seen==0) begin
                    address_byte=incoming;
                    if(incoming!==8'h4e)
                        $fatal(1,"PCF8574 address expected 4E got %02h",incoming);
                end else if(bytes_seen==1) begin
                    pcf_byte=incoming;
                    if(writes_seen>=72) $fatal(1,"extra PCF writes");
                    if(pcf_byte!==expected[writes_seen])
                        $fatal(1,"PCF[%0d] expected=%02h got=%02h",
                            writes_seen,expected[writes_seen],pcf_byte);
                    writes_seen=writes_seen+1;
                end else $fatal(1,"more than two data bytes in transaction");
                bytes_seen=bytes_seen+1;
                bit_pos=0; incoming=0;
            end
        end
    end
    always @(posedge sda) begin
        if(resetn && scl && bytes_seen==2) begin
            transactions=transactions+1;
        end
    end
    // Typical PCF8574 backpack value for LCD nibble:
    // {D7..D4, BL=1, E, RW=0, RS}.
    function [7:0] io_val;
        input [3:0] nib;
        input e;
        input rs;
        begin io_val={nib,1'b1,e,1'b0,rs}; end
    endfunction
    task add_nib;
        input [3:0] nib;
        input rs;

// Separate NACK regression: without a responding I2C slave,
// both address and data ACK slots remain HIGH from the pull-up.
module tb_i2c_write_nack;
    reg clk=0;always #5 clk=~clk;
    reg resetn=0,cmd_valid=0;
    wire cmd_ready,scl_drive_low,sda_drive_low,done_pulse,nack_pulse;
    wire sda = sda_drive_low ? 1'b0 : 1'b1;
    integer seen_done=0,seen_nack=0,clock_rises=0;
    wire bus_scl = scl_drive_low ? 1'b0 : 1'b1;
    // (변경) STOP은 SCL을 1회 추가 상승시키므로
    // 데이터 및 ACK 구간의 SCL 상승만 카운트한다.
    always @(posedge bus_scl) if(resetn &&
        (dut.state==4'd4 || dut.state==4'd7))
        clock_rises=clock_rises+1;
    i2c_write_byte_master #(.CLK_DIV(3)) dut(
        .clk(clk),.resetn(resetn),.cmd_valid(cmd_valid),
        .cmd_ready(cmd_ready),.cmd_addr(7'h27),.cmd_data(8'h08),
        .sda_in(sda),.scl_drive_low(scl_drive_low),
        .sda_drive_low(sda_drive_low),.done_pulse(done_pulse),
        .nack_pulse(nack_pulse)
    );
    always @(posedge clk) if(done_pulse) begin
        seen_done=seen_done+1;
        if(!nack_pulse) $fatal(1,"NACK not reported");
        seen_nack=seen_nack+1;
    end
    initial begin
        repeat(3) @(negedge clk);resetn=1;
        @(negedge clk);cmd_valid=1;
        @(negedge clk);cmd_valid=0;
        wait(done_pulse);repeat(3) @(negedge clk);
        if(seen_done!=1 || seen_nack!=1 || clock_rises!=9 ||
           !cmd_ready || scl_drive_low || sda_drive_low)
            $fatal(1,"NACK completion/STOP/idle failure clocks=%0d done=%0d nack=%0d",
                   clock_rises,seen_done,seen_nack);
        $display("P2_14_I2C_NACK_HANDLING_PASS");
        $finish;
    end
    initial begin #200000; $fatal(1,"TIMEOUT");end
endmodule
