00과제 개요

서보 모터는 PWM(펄스 폭 변조)의 듀티 비율보다 HIGH가 지속되는 펄스 폭이 중요하다. 기존 P2-04의 PWM RTL·AXI IP를 수정하지 않고 새 IP 인스턴스를 추가하여 SG90을 제어한다. 목표는 각도 0°, 90°, 180° 명령을 20ms PWM 프레임에 대응시키는 것이다.

Vivado 합성, Vitis 타깃 빌드, 전기 연결 및 Basys3 실물 시험은 아직 수행되지 않았다. 아래 주파수·핀·각도는 실제 보드에서 검증해야 한다.

01전체 시스템 구조

기존 pwm_core.v와 p2_04_axi_pwm_v1_0*.v를 변경 없이 재사용한다. Block Design(블록 디자인)에 동일 IP의 두 번째 인스턴스를 추가하고, 신규 출력만 서보 신호선으로 보낸다.

02펄스폭 계산

각도 명령HIGH 펄스폭100MHz 주기 카운트100MHz HIGH 카운트
0°1,000μs2,000,000100,000
90°1,500μs2,000,000150,000
180°2,000μs2,000,000200,000

환산식: period_ticks = clock_hz × 20,000 / 1,000,000, duty_ticks = clock_hz × pulse_us / 1,000,000. 여기서 각도의 증가는 펄스폭의 선형 변화로 정의한 교육용 명령 값이다. 실제 축 각도는 기구 오차와 호환 모터 편차가 있으므로 각도계로 확인한다.

04장 PWM Core는 설정을 프레임 경계에서 반영하므로 고주파 글리치 없이 다음 주기부터 새 펄스폭을 적용한다.

Basys3 I/O에서 모터 전원을 공급하거나 5V를 입력으로 넣지 않는다. 전원 투입 전에 서보 혼을 분리하고 무부하 중립 위치부터 확인한다. 모터가 끝단에 걸리거나 떨리면 즉시 전원을 차단한다. 실제 SIG 핀은 기존 Pmod 배정 및 XDC와 충돌하지 않는 핀을 확인한 뒤 지정한다.

04Vivado 실습 절차

  1. 제2부 10장의 실제 성공·진행 프로젝트를 File → Project → Save As로 별도 soc_ip_design_11에 저장한다. 기존 원본 경로/소스를 변경하지 않는다.
  2. 04장의 p2_04_axi_pwm_v1_0 사용자 IP가 현행 프로젝트 IP Repository에 등록되어 있는지 확인한다. 없는 경우 04장 RTL 원본을 별도 복제 폴더에서 패키징한다.
  3. Block Design에 동일 PWM IP의 신규 인스턴스를 추가한다. 기존 PWM IP는 LED1 소유권을 유지한다.
  4. 새 IP의 s00_axi_aclk와 s00_axi_aresetn을 기존 AXI 클록·리셋 체계에 연결한다. 실제 주파수를 Block Design에서 확인한다.
  5. AXI Interconnect 연결 및 Address Editor 신규 주소를 배정한다. 기존 주소를 임의로 재할당하지 않는다.
  6. 신규 IP의 pwm_led_o를 외부로 내보내고 출력 포트 이름을 servo_pwm_o로 정한다. pwm_led_o는 IP 내부 포트 이름이므로 변경할 필요가 없다.
  7. Validate Design → Generate Output Products → Create HDL Wrapper. 사용자 Top은 servo_pwm_o 출력 포트와 생성 Wrapper의 포트 단순 연결만 추가한다.
  8. 새 출력용 XDC 핀과 LVCMOS33을 실제 사용 가능한 Pmod 핀 기준으로 지정한다. 기존 모터·센서 포트와 충돌 여부를 확인한다.
  9. Run Synthesis → Run Implementation → Timing Summary를 확인한 뒤 Bitstream 및 XSA를 생성한다.

Top에서 제어용 산술, 비교, always, PWM 생성 회로를 직접 작성하지 않는다.

sw/servo_logic.c · C · 소스 원문
#include "servo_logic.h"
#include <limits.h>
// SG90 초기 교육 범위: 20ms 프레임, 1~2ms HIGH 펄스.
// 제조사·실물 차이로 실제 기계적 0/180도와 정확히 일치하지 않을 수 있다.
int servo_angle_to_config(uint32_t clock_hz, int angle_deg,
                          servo_config_t *out)
{
    uint64_t period, pulse_us, duty;
    if (!out || clock_hz < 1000000u || clock_hz > 200000000u ||
        angle_deg < 0 || angle_deg > 180)
        return -1;
    // round-to-nearest us: 0도 1000us, 90도 1500us, 180도 2000us.
    // 정수 연산만 사용. 32비트 곱셈 오버플로 방지를 위해 uint64_t 사용.
    pulse_us = 1000u + (uint64_t)(angle_deg * 1000 + 90) / 180u;
    period = ((uint64_t)clock_hz * 20000u + 500000u) / 1000000u;
    duty = ((uint64_t)clock_hz * pulse_us + 500000u) / 1000000u;
    if (period < 2u || period > UINT32_MAX || duty >= period)
        return -1;
    out->period_ticks = (uint32_t)period;
    out->duty_ticks = (uint32_t)duty;
    return 0;
}

sw/servo_axi.c · C · 소스 원문
#include "servo_axi.h"
#include "servo_logic.h"
#include "xil_io.h"
// P2-04 원본 AXI PWM IP의 변경 없는 오프셋 사용.
// 0x00 PERIOD_CFG / 0x04 DUTY_CFG / 0x08 CONTROL / 0x0C APPLY
// 0x10 STATUS / 0x14 ACTIVE_PERIOD / 0x18 ACTIVE_DUTY / 0x1C ID
#define PWM_PERIOD  0x00u
#define PWM_DUTY    0x04u
#define PWM_CONTROL 0x08u
#define PWM_APPLY   0x0Cu
int servo_axi_set_angle(uintptr_t baseaddr, uint32_t clock_hz, int angle_deg)
{
    servo_config_t c;
    if (servo_angle_to_config(clock_hz, angle_deg, &c) != 0)
        return -1;
    // 별도 P2-11 인스턴스에만 접근. 기존 LED1 PWM IP 주소 사용 금지.
    Xil_Out32(baseaddr + PWM_PERIOD, c.period_ticks);
    Xil_Out32(baseaddr + PWM_DUTY, c.duty_ticks);
    Xil_Out32(baseaddr + PWM_CONTROL, 1u); // enable=1, normal polarity
    Xil_Out32(baseaddr + PWM_APPLY, 1u);   // 다음 PWM 주기 경계 적용
    return 0;
}
int servo_axi_disable(uintptr_t baseaddr)
{
    // 모터 신호를 바로 LOW로 만들지 않고 다음 프레임 경계에 적용.
    Xil_Out32(baseaddr + PWM_CONTROL, 0u);
    Xil_Out32(baseaddr + PWM_APPLY, 1u);
    return 0;
}

tests/tb_servo_pwm_reuse.v — 전체 코드 열기
tests/tb_servo_pwm_reuse.v · VERILOG · 소스 원문
\`timescale 1ns/1ps
// P2-11: P2-04 pwm_core.v를 수정 없이 재사용하는 PWM 경계 동작 테스트.
// 20ms 실제 주기를 200 tick으로 축척한 교육용 시뮬레이션.
// 실물 설정(100MHz)은 C 함수 단위 테스트에서 별도로 검증.
module tb_servo_pwm_reuse;
    reg clk = 1'b0;
    always #5 clk = ~clk;
    reg resetn=0, apply=0, enable=0;
    reg [31:0] period=200, duty=10;
    wire pwm;
    wire pending;
    wire [31:0] active_period, active_duty;
    wire active_enable, active_polarity;
    integer highs, i;
    pwm_core #(.RESET_PERIOD(32'd200)) dut (
        .clk(clk), .resetn(resetn), .cfg_apply(apply),
        .cfg_period(period), .cfg_duty(duty),
        .cfg_enable(enable), .cfg_polarity(1'b0),
        .pwm_out(pwm), .pending(pending),
        .active_period(active_period), .active_duty(active_duty),
        .active_enable(active_enable), .active_polarity(active_polarity)
    );
    task request_pwm;
        input [31:0] new_duty;
        begin
            @(negedge clk);
            duty=new_duty; enable=1; apply=1;
            @(negedge clk); apply=0;
            // 설정은 주기 경계에서만 활성화된다.
            wait(active_enable && active_duty == new_duty);
        end
    endtask
    task check_frame;
        input integer expected_high;
        begin
            highs=0;
            // 주기 시작을 동기화하기 위해 high-to-low 후 다음 high 시작 검출.
            wait(pwm == 1'b0);
            @(posedge pwm);
            for (i=0;i<200;i=i+1) begin
                @(negedge clk);
                if (pwm) highs=highs+1;
            end
            if (highs != expected_high) begin
                $display("FAIL expected=%0d actual=%0d",expected_high,highs);
                $fatal(1);
            end
        end
    endtask
    initial begin
        repeat(4) @(negedge clk);
        resetn=1;
        request_pwm(10);
        check_frame(10);
        request_pwm(15);
        check_frame(15);
        request_pwm(20);
        check_frame(20);
        $display("P2_11_RTL_PWM_REUSE PASS");
        $finish;
    end
    initial begin #200000; $fatal(1,"TIMEOUT"); end
endmodule

sw/servo_command.c — 전체 코드 열기
sw/servo_command.c · C · 소스 원문
#include "servo_command.h"
#include <string.h>
// 기존 P2-10 UART Parser는 유지한다. SERVO 명령 전용 독립 Parser.
// 엄격한 명령 규격: "SERVO 0"~"SERVO 180", "SERVO OFF".
// 뒤쪽 CR/LF는 기존 Line Receiver에서 제거해야 한다.
int servo_parse_command(const char *line, servo_command_t *cmd)
{
    int deg = 0;
    const char *p;
    if (!line || !cmd) return -1;
    if (strncmp(line, "SERVO", 5) != 0 ||
        (line[5] != '\0' && line[5] != ' '))
        return 0;
    if (strcmp(line, "SERVO OFF") == 0) {
        cmd->kind = SERVO_CMD_OFF;
        cmd->angle_deg = 0;
        return 1;
    }
    if (strncmp(line, "SERVO ", 6) != 0) return -1;
    p = line + 6;
    if (*p == '\0') return -1;
    while (*p != '\0') {
        if (*p < '0' || *p > '9') return -1;
        deg = deg * 10 + (*p - '0');
        if (deg > 180) return -1; // 큰 정수 입력도 안전하게 중단
        ++p;
    }
    cmd->kind = SERVO_CMD_ANGLE;
    cmd->angle_deg = deg;
    return 1;
}
