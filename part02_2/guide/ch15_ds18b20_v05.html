00개요

DS18B20(디에스18비20)은 1-Wire(원와이어, 단일 데이터선 통신) 디지털 온도 센서다. I²C처럼 클록선이 따로 없고, 하나의 DQ 데이터선을 LOW로 당겼다가 놓는 동작의 시간을 측정하여 리셋·명령·데이터를 전달한다.

최초 실습은 외부 3.3V 전원을 사용하는 센서 한 개를 대상으로 한다. Reset/Presence, Skip ROM, Convert T, Read Scratchpad, 9바이트 CRC 검증까지 단계적으로 실행한다.

실행 결과는 모의 센서에 대한 Icarus Verilog RTL 시뮬레이션이다. Vivado 합성/타이밍, Vitis 실제 MicroBlaze V 타깃, DS18B20 실물 온도 측정은 현 단계에서 수행하지 않았다. 시뮬레이션 PASS가 실제 센서 PASS를 의미하지 않는다.

01시스템 계층형 구조도

컨트롤러는 센서 명령·데이터 검증, 슬롯 엔진은 시간 규칙, IOBUF는 실제 단일선의 0/High-Z 핀 동작을 전담한다. Top은 핀 연결만 수행한다.

모든 명령 바이트와 데이터 바이트는 LSB First(최하위 비트부터 전송) 방식이다. 앞선 SPI와 I²C의 MSB First 흐름과 구분해야 한다. Analog Devices DS18B20 공식 데이터시트.

04온도 데이터 형식과 계산

Raw(16비트)의미온도(°C)
0x0191부호 있는 정수 401, 단위 1/16°C+25.0625
0x0000정수 00.0000
0xFF5E16비트 signed 값 −162−10.1250

12비트 모드에서 한 계수의 해상도는 0.0625°C이다. Scratchpad 0번 바이트는 온도의 하위 8비트, 1번 바이트는 상위 8비트다. 단순히 16비트 unsigned 숫자로 표시하지 않고 signed 값으로 해석해야 한다.

CRC-8 다항식은 x⁸+x⁵+x⁴+1이고, LSB-first 구현은 비트 이동과 XOR 상수 0x8C를 활용한다. 9번째 바이트 CRC와 일치할 때에만 공개 온도 데이터를 갱신한다.

05DS18B20 배선 및 안전

센서 핀Basys3 기준주의
VDD3.3V 외부전원 공급부품 버전·핀 배열 확인
GNDBasys3 GND와 공통배선 전 전원 OFF
DQFPGA I/O 1개, Open-drain약 4.7kΩ을 3.3V에 풀업

Basys3 FPGA 신호선에 5V를 연결하지 않는다. 전원 없이 DQ로 센서에 전류를 공급하는 기생전원 모드는 이 RTL에서 지원하지 않으며 별도 Strong Pullup 설계가 필요하다. XDC 핀은 현행 프로젝트의 기존 센서·서보·모터 핀 점유 상태를 확인할 때 확정한다.

rtl/onewire_slot_master.v — 전체 코드 열기
rtl/onewire_slot_master.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// P2-15 1-Wire slot engine: reset/presence, write bit, read bit.
// Single system clk, open-drain dq_drive_low (1=LOW; 0=Hi-Z).
// CLK_HZ must be an integer multiple of 1 MHz.
module onewire_slot_master #(
    parameter integer CLK_HZ=100000000
)(
    input wire clk,input wire resetn,
    input wire cmd_valid,output wire cmd_ready,
    input wire [1:0] cmd_op, // 0=RESET, 1=WRITE, 2=READ
    input wire cmd_bit,
    input wire dq_in,output reg dq_drive_low,
    output reg done_pulse,output reg presence,
    output reg read_bit
);
    localparam integer CYCLES_PER_US=CLK_HZ/1000000;
    localparam [2:0] IDLE=0, RESET_LOW=1, PRES_WAIT=2,
        PRES_RECOVER=3, SLOT_LOW=4, READ_WAIT=5,
        SLOT_RELEASE=6, FINISH=7;
    reg [2:0] state;
    reg [31:0] us_div;
    reg [15:0] remain;
    reg [1:0] op_q;
    reg bit_q;
    wire tick_us = (us_div == CYCLES_PER_US-1);
    assign cmd_ready=(state==IDLE);
    always @(posedge clk) begin
        if(!resetn) begin
            state<=IDLE;us_div<=0;remain<=0;
            op_q<=0;bit_q<=0;dq_drive_low<=0;
            done_pulse<=0;presence<=0;read_bit<=1;
        end else begin
            done_pulse<=0;
            if(state==IDLE) us_div<=0;
            else if(tick_us) us_div<=0;
            else us_div<=us_div+1;
            if(state==IDLE) begin
                dq_drive_low<=0;
                if(cmd_valid) begin
                    op_q<=cmd_op;bit_q<=cmd_bit;
                    if(cmd_op==0) begin
                        presence<=0;dq_drive_low<=1;
                        remain<=16'd480;state<=RESET_LOW;
                    end else if(cmd_op==1 || cmd_op==2) begin
                        dq_drive_low<=1;
                        remain<= (cmd_op==2 || cmd_bit) ? 16'd6 : 16'd60;
                        state<=SLOT_LOW;
                    end else begin
                        state<=FINISH;
                    end
                end
            end else if(tick_us) begin
                if(remain>1) remain<=remain-1;
                else begin
                    case(state)
                    RESET_LOW: begin
                        dq_drive_low<=0;remain<=16'd70;
                        state<=PRES_WAIT;
                    end
                    PRES_WAIT: begin
                        presence<=~dq_in;remain<=16'd410;
                        state<=PRES_RECOVER;
                    end
                    PRES_RECOVER: state<=FINISH;
                    SLOT_LOW: begin
                        dq_drive_low<=0;
                        if(op_q==2) begin
                            remain<=16'd9;state<=READ_WAIT;
                        end else begin
                            remain<=bit_q ? 16'd64 : 16'd10;
                            state<=SLOT_RELEASE;
                        end
                    end
                    READ_WAIT: begin
                        read_bit<=dq_in;
                        remain<=16'd55;state<=SLOT_RELEASE;
                    end
                    SLOT_RELEASE: state<=FINISH;
                    FINISH: state<=IDLE;
                    default: state<=IDLE;
                    endcase
                end
            end
            if(state==FINISH && tick_us) begin
                done_pulse<=1;
                dq_drive_low<=0;
            end
        end
    end
endmodule
rtl/ds18b20_controller.v — 전체 코드 열기
rtl/ds18b20_controller.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// DS18B20 one externally-powered device on bus (VDD, DQ, GND).
// Reset -> Skip ROM(CC) -> Convert T(44) -> 750ms wait
// -> Reset -> Skip ROM(CC) -> Read Scratchpad(BE) -> 9 bytes -> CRC8.
// No strong-pullup/parasite-power support, multi-drop addressing,
// variable resolution, interrupt, or AXI interface in this version.
module ds18b20_controller #(
    parameter integer CLK_HZ=100000000,
    parameter integer CONVERT_WAIT_US=750000
)(
    input wire clk,input wire resetn,input wire start_pulse,
    input wire dq_in,output wire dq_drive_low,
    output reg busy,output reg done_pulse,output reg error_pulse,
    output reg [2:0] error_code, // 1=missing presence, 2=CRC
    output reg signed [15:0] temperature_raw
);
    localparam [3:0] IDLE=0,ISSUE=1,WAIT_OP=2,
        CHECK_RESET=3,WRITE_ADV=4,WAIT_CONVERT=5,
        READ_ADV=6,COMPLETE=7,FAULT=8;
    reg [3:0] state;
    // phases: reset0, CC, 44, wait, reset1, CC, BE, read9
    reg [2:0] phase;
    reg [7:0] tx_data;
    reg [3:0] bit_index;
    reg [3:0] byte_index;
    reg [7:0] scratchpad_byte;
    reg [7:0] crc;
    reg [7:0] low_temp;
    reg [7:0] high_temp;
    reg [31:0] conv_div;
    reg [31:0] conv_remaining;
    reg cmd_valid;
    reg [1:0] cmd_op;
    reg cmd_bit;
    wire cmd_ready,slot_done,presence,read_bit;
    onewire_slot_master #(.CLK_HZ(CLK_HZ)) u_slot (
        .clk(clk),.resetn(resetn),.cmd_valid(cmd_valid),
        .cmd_ready(cmd_ready),.cmd_op(cmd_op),.cmd_bit(cmd_bit),
        .dq_in(dq_in),.dq_drive_low(dq_drive_low),
        .done_pulse(slot_done),.presence(presence),.read_bit(read_bit)
    );
    function [7:0] crc_next;
        input [7:0] old_crc;
        input bit_value;
        reg feedback;
        begin
            feedback=old_crc[0]^bit_value;
            crc_next={1'b0,old_crc[7:1]} ^ (feedback ? 8'h8c : 8'h00);
        end
    endfunction
    always @(posedge clk) begin
        if(!resetn) begin
            state<=IDLE;phase<=0;tx_data<=0;bit_index<=0;
            byte_index<=0;scratchpad_byte<=0;crc<=0;low_temp<=0;high_temp<=0;
            conv_div<=0;conv_remaining<=0;
            cmd_valid<=0;cmd_op<=0;cmd_bit<=0;
            busy<=0;done_pulse<=0;error_pulse<=0;
            error_code<=0;temperature_raw<=0;
        end else begin
            done_pulse<=0;error_pulse<=0;cmd_valid<=0;
            case(state)
            IDLE: if(start_pulse) begin
                busy<=1;error_code<=0;phase<=0;
                bit_index<=0;byte_index<=0;crc<=0;
                state<=ISSUE;
            end
            ISSUE: if(cmd_ready) begin
                cmd_valid<=1;
                if(phase==0 || phase==4) begin
                    cmd_op<=0;cmd_bit<=0;
                end else if(phase==7) begin
                    cmd_op<=2;cmd_bit<=0;
                end else begin
                    cmd_op<=1;cmd_bit<=tx_data[bit_index];
                end
                state<=WAIT_OP;
            end
            WAIT_OP: if(slot_done) begin
                if(phase==0 || phase==4) state<=CHECK_RESET;
                else if(phase==7) state<=READ_ADV;
                else state<=WRITE_ADV;
            end
            CHECK_RESET: begin
                if(!presence) begin
                    error_code<=1;state<=FAULT;
                end else begin
                    if(phase==0) begin phase<=1;tx_data<=8'hcc;end
                    else begin phase<=5;tx_data<=8'hcc;end
                    bit_index<=0;state<=ISSUE;
                end
            end
            WRITE_ADV: begin
                if(bit_index!=7) begin
                    bit_index<=bit_index+1;state<=ISSUE;
                end else begin
                    bit_index<=0;
                    case(phase)
                    1: begin phase<=2;tx_data<=8'h44;state<=ISSUE;end
                    2: begin
                        phase<=3;conv_div<=0;
                        conv_remaining<=CONVERT_WAIT_US;
                        state<=WAIT_CONVERT;
                    end
                    5: begin phase<=6;tx_data<=8'hbe;state<=ISSUE;end
                    6: begin
                        phase<=7;byte_index<=0;crc<=0;
                        scratchpad_byte<=0;state<=ISSUE;
                    end
                    default: state<=FAULT;
                    endcase
                end
            end
            WAIT_CONVERT: begin
                if(conv_remaining==0) begin
                    phase<=4;state<=ISSUE;
                end else if(conv_div==CLK_HZ/1000000-1) begin
                    conv_div<=0;
                    conv_remaining<=conv_remaining-1;
                end else conv_div<=conv_div+1;
            end
            READ_ADV: begin
                scratchpad_byte[bit_index]<=read_bit;
                if(byte_index<8) crc<=crc_next(crc,read_bit);
                if(bit_index!=7) begin
                    bit_index<=bit_index+1;state<=ISSUE;
                end else begin
                    bit_index<=0;
                    if(byte_index==0) low_temp<={read_bit,scratchpad_byte[6:0]};
                    if(byte_index==1)
                        high_temp<={read_bit,scratchpad_byte[6:0]};
                    if(byte_index==8) begin
                        if({read_bit,scratchpad_byte[6:0]}==crc) begin
                            // (변경) CRC 정상일 때에만 공개 온도 레지스터 갱신.
                            temperature_raw<={high_temp,low_temp};
                            state<=COMPLETE;
                        end
                        else begin
                            error_code<=2;state<=FAULT;
                        end
                    end else begin
                        byte_index<=byte_index+1;
                        scratchpad_byte<=0;state<=ISSUE;
                    end
                end
            end
            COMPLETE: begin busy<=0;done_pulse<=1;state<=IDLE;end
            FAULT: begin busy<=0;error_pulse<=1;state<=IDLE;end
            default: state<=FAULT;
            endcase
        end
    end
endmodule

tests/tb_ds18b20_controller.v — 전체 코드 열기
tests/tb_ds18b20_controller.v · VERILOG · 소스 원문
`timescale 1ns/1ps
// P2-15 self-checking simulated DS18B20, does NOT model analog bus.
// Checks 2 resets/presence, CC/44/CC/BE TX, nine scratchpad bytes,
// signed temperature_raw, CRC rejection, missing presence.
module tb_ds18b20_controller;
  reg clk=0;always #5 clk=~clk;
  reg resetn=0,start=0,presence_enabled=1,corrupt_crc=0;
  wire dq_drive_low,done,error,busy;
  wire [2:0] error_code;
  wire signed [15:0] temperature_raw;
  reg [7:0] scratch[0:8];
  reg [7:0] crc_local;
  integer i,j,case_number=0,command_bits=0,reset_count=0;
  reg [7:0] command_bytes[0:3];
  reg [7:0] current_byte;
  reg slave_low;
  wire dq_in = !(dq_drive_low || slave_low);
  ds18b20_controller #(
      .CLK_HZ(1000000),.CONVERT_WAIT_US(20)
  ) dut(
    .clk(clk),.resetn(resetn),.start_pulse(start),
    .dq_in(dq_in),.dq_drive_low(dq_drive_low),
    .busy(busy),.done_pulse(done),.error_pulse(error),
    .error_code(error_code),.temperature_raw(temperature_raw)
  );
  function [7:0] crc_byte;
    input [7:0] crc_in;
    input [7:0] b;
    reg [7:0] c;
    integer n;
    begin
      c=crc_in;
      for(n=0;n<8;n=n+1) begin
        if(c[0]^b[n]) c=(c>>1)^8'h8c;
        else c=c>>1;
      end
      crc_byte=c;
    end
  endfunction
  task set_scratchpad;
    input [15:0] raw;
    begin
      scratch[0]=raw[7:0];scratch[1]=raw[15:8];
      scratch[2]=8'h4b;scratch[3]=8'h46;
      scratch[4]=8'h7f;scratch[5]=8'hff;
      scratch[6]=8'h0c;scratch[7]=8'h10;
      crc_local=0;
      for(i=0;i<8;i=i+1)
        crc_local=crc_byte(crc_local,scratch[i]);
      scratch[8]=crc_local^(corrupt_crc ? 8'h01 : 8'h00);
    end
  endtask
  // Simplified digital slave drives ACK-like presence and read-zero bits
  // at the exact sampling state. This is a functional bus model, not a
  // nanosecond jitter/timing compliance test.
  always @(*) begin
    slave_low=0;
    if(presence_enabled && dut.u_slot.state==3'd2)
      slave_low=1;
    if(dut.u_slot.state==3'd5 && dut.phase==7 &&
       !scratch[dut.byte_index][dut.bit_index])
      slave_low=1;
  end

  // Collect commands only at slot transaction start, avoiding races
  // between testbench and core's registered data pins.
  always @(posedge clk) begin
    if(resetn && dut.u_slot.cmd_valid && dut.u_slot.cmd_ready) begin
      if(dut.u_slot.cmd_op==0) begin
        reset_count=reset_count+1;
      end else if(dut.u_slot.cmd_op==1) begin
        current_byte[command_bits%8]=dut.u_slot.cmd_bit;
        command_bits=command_bits+1;
        if(command_bits%8==0)
          command_bytes[command_bits/8-1]=current_byte;
      end
    end
  end
  task reset_test;
    begin
      resetn=0;start=0;
      repeat(4) @(negedge clk);
      resetn=1;command_bits=0;reset_count=0;
    end
  endtask
  task kick;
    begin
      @(negedge clk);start=1;
      @(negedge clk);start=0;
    end
  endtask
  initial begin
    corrupt_crc=0;set_scratchpad(16'h0191);
    reset_test; kick;
    wait(done); #1;
    if(error || error_code || busy || temperature_raw!==16'h0191)
      $fatal(1,"positive temp incorrect: %h",temperature_raw);
    if(reset_count!=2 || command_bits!=32 ||
       command_bytes[0]!==8'hcc || command_bytes[1]!==8'h44 ||
       command_bytes[2]!==8'hcc || command_bytes[3]!==8'hbe)
      $fatal(1,"command sequence wrong resets=%0d bits=%0d %h %h %h %h",
             reset_count,command_bits,
             command_bytes[0],command_bytes[1],
             command_bytes[2],command_bytes[3]);
    $display("P2_15_DS18B20_POSITIVE_CRC_COMMAND_PASS");

    corrupt_crc=0;set_scratchpad(16'hff5e);
    reset_test;kick;wait(done);#1;
    if(busy || temperature_raw!==16'hff5e)
      $fatal(1,"negative signed temperature wrong: %h",temperature_raw);
    $display("P2_15_DS18B20_NEGATIVE_TEMP_PASS");

    corrupt_crc=1;set_scratchpad(16'h0191);
    reset_test;kick;wait(error);#1;
    if(error_code!=2 || busy || temperature_raw!==16'h0000)
      $fatal(1,"CRC error not rejected or untrusted temperature published");
    $display("P2_15_DS18B20_CRC_REJECT_PASS");

    corrupt_crc=0;set_scratchpad(16'h0191);presence_enabled=0;
    reset_test;kick;wait(error);#1;
    if(error_code!=1 || reset_count!=1 || busy)
      $fatal(1,"missing presence not detected");
    $display("P2_15_DS18B20_NO_PRESENCE_PASS");

    $display("P2_15_DS18B20_CONTROLLER_RTL PASS");
    $finish;
  end
  initial begin #4000000;$fatal(1,"TIMEOUT");end
endmodule
